local lastDropDist = nil

local function findDropAtAnchor()
    local best, bestD = nil, CFG.Radius
    local function rec(o, d)
        if d > 14 then return end
        for _, c in ipairs(o:GetChildren()) do
            if c:IsA("BasePart") and c.Name:match(UUID_PAT) then
                local ok, dist = pcall(function()
                    return (c.Position - CFG.Anchor).Magnitude
                end)
                if ok and dist <= bestD then
                    best, bestD = c, dist
                end
            end
            rec(c, d + 1)
        end
    end
    rec(Workspace, 0)
    lastDropDist = bestD
    return best
end

local function findStage()
    local p = Workspace:FindFirstChild(CFG.StagePart)
    if p and p:IsA("BasePart") then return p end
    local function rec(o, d)
        if d > 14 then return nil end
        for _, c in ipairs(o:GetChildren()) do
            if c.Name == CFG.StagePart and c:IsA("BasePart") then return c end
            local r = rec(c, d + 1)
            if r then return r end
        end
        return nil
    end
    return rec(Workspace, 0)
end

local function newSlider(parent, name, y, min, max, step, init, fmt, onSet)
    local holder = Instance.new("Frame")
    holder.Name = "sl_" .. name
    holder.BackgroundTransparency = 1
    holder.Size = UDim2.new(1, -24, 0, 34)
    holder.Position = UDim2.fromOffset(12, y)
    holder.Parent = parent

    newLabel(holder, "lab", UDim2.new(0.5, 0, 0, 14), UDim2.fromOffset(0, 0), name, 9, DIM,
        Enum.TextXAlignment.Left, Enum.Font.GothamSemibold)
    local val = newLabel(holder, "val", UDim2.new(0.5, 0, 0, 14), UDim2.new(0.5, 0, 0, 0), "", 9, TXT,
        Enum.TextXAlignment.Right, Enum.Font.GothamSemibold)

    local track = newFrame(holder, "track", UDim2.new(1, 0, 0, 6), UDim2.new(0, 0, 1, -6),
        Color3.fromRGB(40, 40, 54), 3)
    local fill = newFrame(track, "fill", UDim2.new(0, 0, 1, 0), UDim2.fromOffset(0, 0), VIOLET, 3)
    local sknob = newFrame(track, "knob", UDim2.fromOffset(12, 12), UDim2.new(0, -6, 0.5, -6),
        Color3.fromRGB(240, 240, 246), 6)

    local cur = init
    local function apply(v, fire)
        v = math.clamp(v, min, max)
        v = math.floor(v / step + 0.5) * step
        cur = v
        local frac = (max > min) and (v - min) / (max - min) or 0
        fill.Size = UDim2.new(frac, 0, 1, 0)
        sknob.Position = UDim2.new(frac, -6, 0.5, -6)
        val.Text = fmt(v)
        if fire then onSet(v) end
    end

    local dragging = false
    track.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            local frac = math.clamp((i.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
            apply(min + frac * (max - min), true)
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement
                     or i.UserInputType == Enum.UserInputType.Touch) then
            local frac = math.clamp((i.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
            apply(min + frac * (max - min), true)
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    apply(init, false)
    return { get = function() return cur end }
end

local phaseTotal = 1

local function updateStatus()
    local bodyText
    if not enabled then
        bodyText = "off"
    elseif phase == "cool" then
        bodyText = ("cool %.1fs"):format(math.max(waitLeft, 0))
    elseif phase == "stage" then
        bodyText = "arming " .. CFG.StagePart .. "…"
    elseif phase == "wait" then
        bodyText = ("wait %.1fs"):format(math.max(waitLeft, 0))
    elseif phase == "farm" then
        bodyText = farmInfo
    elseif phase == "nostage" then
        bodyText = CFG.StagePart .. " missing"
    else
        bodyText = "…"
    end
    statusLab.Text = bodyText
    statusLab.TextColor3 = enabled and TXT or DIM
end

local function setEnabled(v)
    enabled = v
    tween(switch, { BackgroundColor3 = v and GREEN or RED })

    tween(orbDot, { BackgroundColor3 = v and GREEN or RED })
    log(v and "toggled ON" or "toggled OFF")
    updateStatus()
end

task.spawn(function()
    local firstCycle  = true
    local trackedDrop = nil
    while true do
        if not enabled then
            phase = "idle"
            firstCycle = true
            trackedDrop = nil
            task.wait(0.25)
        else
            local char = LocalPlayer.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")
            local hum  = char and char:FindFirstChildOfClass("Humanoid")
            if not root or not hum or hum.Health <= 0 then
                phase = "idle"
                task.wait(1)
            else
                if not firstCycle then
                    phase = "cool"
                    local c = CFG.CoolDelayMs * (1 + (math.random() * 2 - 1) * CFG.CoolJitterPct)
                    phaseTotal = c / 1000
                    log(("cool %.2fs"):format(c / 1000))
                    local dl = os.clock() + c / 1000
                    while enabled and os.clock() < dl do
                        waitLeft = dl - os.clock()
                        task.wait(0.2)
                    end
                end
                firstCycle = false

                if enabled then
                    cycleN = cycleN + 1
                    log(("── cycle %d ──"):format(cycleN))

                    phase = "stage"
                    phaseTotal = 1
                    local s = findStage()
                    if s then
                        pcall(firetouchinterest, s, root, 0)
                        pcall(firetouchinterest, s, root, 1)
                        log("arm " .. CFG.StagePart .. " · single tap sent")
                        phase = "wait"
                    else
                        log("arm FAILED · " .. CFG.StagePart .. " not found")
                        phase = "nostage"
                    end

                    local d = CFG.StageDelayMs * (1 + (math.random() * 2 - 1) * CFG.PadJitterPct)
                    phaseTotal = d / 1000
                    log(("pad wait %.2fs"):format(d / 1000))
                    local deadline = os.clock() + d / 1000
                    while enabled and os.clock() < deadline do
                        waitLeft = deadline - os.clock()
                        task.wait(0.2)
                    end

                    if enabled then
                        phase = "farm"
                        phaseTotal = 1
                        local drop = findDropAtAnchor()
                        if drop then
                            if trackedDrop and trackedDrop ~= drop
                               and not trackedDrop:IsDescendantOf(game) then
                                got = got + 1
                                log("EATEN · got " .. got .. " · " .. trackedDrop.Name .. " vanished")
                            end
                            if drop ~= trackedDrop then
                                log("lock " .. drop.Name .. " @" .. math.floor(lastDropDist + 0.5) .. "st")
                            end
                            trackedDrop = drop
                            farmInfo = ("drop %dst · tapped"):format(math.floor(lastDropDist + 0.5))
                            pcall(firetouchinterest, drop, root, 0)
                            pcall(firetouchinterest, drop, root, 1)
                            log("tap " .. drop.Name .. " · " .. math.floor(lastDropDist + 0.5) .. "st · once")

                            task.wait(1)
                            if trackedDrop and not trackedDrop:IsDescendantOf(game) then
                                got = got + 1
                                log("EATEN · got " .. got .. " · " .. trackedDrop.Name .. " vanished")
                                trackedDrop = nil
                            end
                        else
                            farmInfo = "no drop at anchor"
                            log("farm · no drop at anchor")
                        end
                    end
                end
            end
        end
    end
end)

            frac = (phaseTotal > 0) and math.clamp(waitLeft / phaseTotal, 0, 1) or 0
        elseif phase == "farm" then
            frac = 1
        end
        progFill.Size = UDim2.new(frac, 0, 1, 0)
        progFill.BackgroundColor3 = (phase == "farm") and VIOLET or GREEN
        updateStatus()
    end
