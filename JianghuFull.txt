local function _fkOk(body)
if type(body) ~= "string" or #body < 500 or #body > 350000 then return false end
if not body:find("ForageKit v", 1, true) then return false end
local bodyClean = body:gsub('local FK_SHA256 = "[0-9a-fA-F]+"', 'local FK_SHA256 = nil')
local digest
pcall(function()
if crypt and crypt.hash and crypt.hash.sha256 then digest = crypt.hash.sha256(bodyClean)
elseif syn and syn.crypt and syn.crypt.hash then digest = syn.crypt.hash("sha256", bodyClean)
elseif sha256 then digest = sha256(bodyClean) end
end)
if FK_SHA256 and digest then return digest:lower() == FK_SHA256:lower() end
if FK_SHA256 and not digest then return false end
return true
end

local function ingest(model, found, list)
if found[model] then return end
local rec = tracked[model]
if not rec then
local ok = pcall(function() rec = makeRecord(model) end)
if ok and rec then tracked[model] = rec else return end
else
pcall(function()
rec.root   = model:FindFirstChild("Root", true)
rec.prompt = model:FindFirstChild("HarvestPrompt", true) or model:FindFirstChildOfClass("ProximityPrompt")
if not rec.id then local a = model:GetAttributes(); rec.id = a[CONFIG.HerbAttr] end
end)
end
if rec and rec.id then found[model] = true; list[#list+1] = rec end
end

-- recursive walk: no GetDescendants allocation, yields cooperatively
local function walkScan(root, cap, found, list, counter)
local stack = {root}
while #stack > 0 do
local node = table.remove(stack)
counter[1] = counter[1] + 1
if counter[1] > cap then return true end
local okN, isHerb = pcall(node.GetAttribute, node, CONFIG.HerbAttr)
if okN and isHerb then ingest(node, found, list) end
local okC, children = pcall(node.GetChildren, node)
if okC and children then
for _, child in ipairs(children) do
stack[#stack + 1] = child
end
end
-- yield every 2000 nodes to prevent frame drops
if counter[1] % 2000 == 0 then task.wait() end
end
return false
end

if d < bd then bd = d; best = rec end
end
end
State.nearestByType[id] = best
end

local function groundYAt(x, z, refY)

local hit = Workspace:Raycast(Vector3.new(x, originY, z), Vector3.new(0, -40, 0), _groundParams)
if hit then
local hy = hit.Position.Y
if refY and hy < refY - 10 then return refY end
return hy
end

end
local function groundSnap(pos, refY)
local y = groundYAt(pos.X, pos.Z, refY or pos.Y); if not y then return end
local r = hrp(); if not r then return end
r.CFrame = CFrame.new(pos.X, y + CONFIG.HalfHeight, pos.Z) * (r.CFrame - r.CFrame.Position)

pcall(function() r.AssemblyLinearVelocity = Vector3.zero end)
local h = hum()

end
local function distSqTo(h)
local r = hrp()
local p = h and h.pos and h.pos()
if not (r and p) then return 1e18 end
local d = r.Position - p
return d.X*d.X + d.Z*d.Z
end
local function distTo(h)
local r = hrp()
local p = h and h.pos and h.pos()

end
local function withinRange(h)
local maxD = 8

-- WALK: pure pathfinding along the ground
local _lastPathTarget, _lastPath = nil, nil
local function walkTo(target)
local r, h = hrp(), hum(); if not r or not h then return false end
local saved = h.WalkSpeed
pcall(function() h.WalkSpeed = CONFIG.WalkSpeed end)
pcall(function()
if not _lastPath or not _lastPathTarget or (target - _lastPathTarget).Magnitude > 2 then
local path = PathSvc:CreatePath({AgentRadius = 2, AgentHeight = 5, AgentCanJump = true})
path:ComputeAsync(r.Position, target)
_lastPath = path
_lastPathTarget = target
end
if _lastPath.Status == Enum.PathStatus.Success then
for _, wp in ipairs(_lastPath:GetWaypoints()) do
h:MoveTo(wp.Position)
if wp.Action == Enum.PathWaypointAction.Jump then pcall(function() h.Jump = true end) end
local done = false
local conn = h.MoveToFinished:Connect(function() done = true end)
local deadline = os.clock() + 3
while not done and os.clock() < deadline do task.wait(0.1) end
conn:Disconnect()
if not done then break end
end
else
h:MoveTo(target); task.wait(0.6)
end
end)
pcall(function() h.WalkSpeed = saved end)
return true
end
-- TWEEN: CFrame through obstacles, underground, sky — ignores terrain
local function tweenToTarget(target)
local r = hrp(); if not r then return end
local from = r.CFrame
local to = CFrame.new(target) * (from - from.Position)
local dur = math.max(0.15, (from.Position - to.Position).Magnitude / math.max(1, CONFIG.WalkSpeed))
local tw = TweenSvc:Create(r, TweenInfo.new(dur, Enum.EasingStyle.Linear), {CFrame = to})
tw:Play(); tw.Completed:Wait()
end
-- TELEPORT: instant CFrame, land exactly on the surface the herb sits on
local function teleportToTarget(target)
local r = hrp(); if not r then return end
-- raycast from just above the herb's position (handles underground platforms)
local groundY = groundYAt(target.X, target.Z, target.Y + 4)
if not groundY then groundY = target.Y end
-- teleport to ground + half height (feet exactly on surface)
r.CFrame = CFrame.new(target.X, groundY + CONFIG.HalfHeight, target.Z) * (r.CFrame - r.CFrame.Position)
pcall(function() r.Velocity = Vector3.zero end)
pcall(function() r.AssemblyLinearVelocity = Vector3.zero end)
task.wait(0.05)
local h = hum()
if h then
pcall(function() h:ChangeState(Enum.HumanoidStateType.GettingUp) end)
pcall(function() h:ChangeState(Enum.HumanoidStateType.Running) end)
end
end
-- MODE DISPATCH: each mode has its own approach logic
function Move.approach(h, walkOnly, manual)
local target = h.pos(); local r = hrp()
if not target or not r then return false end
if not State.running and not manual then return false end
local mode = walkOnly and "Walk" or State.mode
if mode == "Walk" then
-- pure pathfinding walk along the ground
walkTo(target); groundSnap(target, target.Y); task.wait(CONFIG.Dwell)
if not withinRange(h) then walkTo(target); groundSnap(target, target.Y); task.wait(0.4) end
return withinRange(h)
end
if mode == "Tween" then
-- tween directly to herb through obstacles, then ground snap before harvest
tweenToTarget(target); task.wait(CONFIG.PostHopSettle)
-- stand on ground at herb position before firing
groundSnap(target, target.Y); task.wait(CONFIG.Dwell)
if not withinRange(h) then
walkTo(target); groundSnap(target, target.Y); task.wait(0.4)
end
return withinRange(h)
end
if mode == "Teleport" then
-- teleport directly to herb, ground snap ensures feet on surface
teleportToTarget(target); task.wait(CONFIG.Dwell)
if not withinRange(h) then
-- if still not in range, walk the last bit
walkTo(target); groundSnap(target, target.Y); task.wait(0.4)
end
return withinRange(h)
end
if mode == "Adaptive" then
local d = distTo(h)
if d > 30 then
-- far: teleport NEAR herb on same surface, then walk the rest
local dir = (target - r.Position).Unit
local nearPoint = target - dir * 12
-- use herb's Y directly (it's on the same platform)
nearPoint = Vector3.new(nearPoint.X, target.Y, nearPoint.Z)
teleportToTarget(nearPoint); task.wait(CONFIG.PostHopSettle)
end
-- walk to herb from current position
walkTo(target); groundSnap(target, target.Y); task.wait(CONFIG.Dwell)
if not withinRange(h) then walkTo(target); groundSnap(target, target.Y); task.wait(0.4) end
return withinRange(h)
end
return false
end

local Harvest = {
lastResult = {},
collected = {},  -- tracks which instanceIds got finish_harvest (collected)
counted = {},    -- instanceIds we already counted as a success (prevents double count)
failAt = {},     -- instanceIds that just failed (retry cooldown, pruned after 30s)
renewKeywords = {"too far","invalid token","renew","expired","token","bound","out of range","not in range"},
}
local function attachHarvestListeners()
pcall(function()
local folder = Replicated:FindFirstChild("Network"); if not folder then return end
for _, o in ipairs(folder:GetDescendants()) do
if o:IsA("RemoteEvent") then
track(o.OnClientEvent:Connect(function(...)
local a = tpack(...)
if not alive then return end
local id = a[1]
-- harvest_result: server tells us the outcome
if o.Name == "harvest_result" then
local ok, msg = a[2], a[3]
if type(id) == "string" then Harvest.lastResult[id] = {ok = ok, msg = msg} end
if ok == false and type(msg) == "string" then
local low = msg:lower()
-- only flag invalid for non-collection errors
if not Harvest.collected[id] then
for _, kw in ipairs(Harvest.renewKeywords) do
if low:find(kw, 1, true) then State.invalid = true; break end
end
end
end
end
-- harvest_started: server accepted our start_harvest, we're in progress
if o.Name == "harvest_started" then
if type(id) == "string" then
-- mark as in-progress (prevents "herb gone" false fail)
Harvest.lastResult[id] = Harvest.lastResult[id] or {ok = nil, msg = "started"}
end
end
-- finish_harvest from server = herb collected, regardless of amount
if o.Name == "finish_harvest" and type(id) == "string" then
Harvest.collected[id] = true
Harvest.lastResult[id] = {ok = true, msg = tostring(a[2] or "collected")}
end
end))
end
end
end)
end
local function viewportArgs(root)
local c = cam()
if not c or not root then return 0.5, 0.5, 20 end
local depth = (c.CFrame.Position - root.Position).Magnitude
if depth < 1 then depth = 1 end
local nx, ny = 0.5, 0.5
pcall(function()
local vp = c:WorldToViewportPoint(root.Position)
local sz = c.ViewportSize
nx = (sz.X > 0 and vp.X / sz.X or 0.5)
ny = (sz.Y > 0 and vp.Y / sz.Y or 0.5)
if not (nx >= 0.03 and nx <= 0.97 and ny >= 0.03 and ny <= 0.97 and vp.Z > 0) then nx, ny = 0.5, 0.5 end
depth = vp.Z > 0 and vp.Z or depth
end)
return nx, ny, depth
end

h.hold = math.clamp(tonumber(attrs.HoldDuration) or h.hold, 0.1, 5)
end

fireServer("Foraging","finish_harvest", h.instanceId, h.token, true, sx, sy, sz)
task.wait(CONFIG.ResultWait)
return true, "remote_pair"
end

local function isTypeLocked(id) return (minMasteryFor(id) or 0) > foragingLevel() end
local function knownSorted() local t = {}; for id in pairs(knownTypes) do t[#t+1] = id end; table.sort(t); return t end
local function nearestInstanceOf(id)
local list = State.byType[id]; if not list then return nil end
local best, bd = nil, 1e9
for _, h in ipairs(list) do
-- skip records whose model was harvested/despawned (stale until next scan)
-- AND skip herbs we already fired finish_harvest on (advance to next herb)
if h.model and h.model.Parent and not Harvest.collected[h.instanceId] then
local fa = Harvest.failAt[h.instanceId]
if not (fa and (os.clock() - fa) < 10) then
local d = distTo(h)
if d < bd then bd = d; best = h end
end
end
end
return best
end
-- returns ALL herbs from selected types, sorted by distance to player (nearest first)
local function selectedHerbsSorted()
local r = hrp(); local p = r and r.Position
if _pickCache and _pickCache.at and p and (_pickCache.at - p).Magnitude <= 2 then
local out, t = {}, _pickCache.list
for i = #t, 1, -1 do
local h = t[i]
local fa = Harvest.failAt[h.instanceId]
if h.model and h.model.Parent and not Harvest.collected[h.instanceId]
and not (fa and (os.clock() - fa) < 10) then
out[#out + 1] = h
end
end
return out
end
local t = {}
for id in pairs(State.selectedTypes) do
if not isTypeLocked(id) then
for _, h in ipairs(State.byType[id] or {}) do
if h.model and h.model.Parent and not Harvest.collected[h.instanceId] then
local fa = Harvest.failAt[h.instanceId]
if not (fa and (os.clock() - fa) < 10) then
local hp = h.pos()
local d = hp and p and ((p.X - hp.X)*(p.X - hp.X) + (p.Z - hp.Z)*(p.Z - hp.Z)) or 1e18
t[#t + 1] = { h = h, d = d }
end
end
end
end
end
table.sort(t, function(a, b) return a.d < b.d end)
local out = {}
for i = 1, #t do out[i] = t[i].h end
_pickCache = { list = out, at = p }
return out
end
-- strategy-aware target selection:
--  "Nearest"  = current behavior — globally nearest herb among selected types
--  "Sequence" = walk selected types in list order (alphabetical, same as the herb list);
--               first type with live herbs wins; within it, always the nearest instance
-- rotation cursor for Sequence mode: index (into the selected-type list) of
-- the type we last picked — the next pick starts AFTER it, so each harvest
-- advances to the next type in the sequence (round-robin, wrap-around)
local seqCursor = 0

-- strategy-aware target selection:
--  "Nearest"  = globally nearest herb among selected types (ignore order)
--  "Sequence" = round-robin across selected types in herb-list order:
--               pick type A's nearest herb -> harvest 1 -> type B's nearest
--               -> type C -> ... -> wrap back to A. Types with no live herbs
--               are skipped. Within each type, always the nearest instance.
local function pickHarvestTarget()
if State.strategy == "Sequence" then
-- ordered list of selected, unlocked types (same order as the herb list)
local types = {}
for _, id in ipairs(knownSorted()) do
if State.selectedTypes[id] and not isTypeLocked(id) and State.byType[id] then
types[#types+1] = id
end
end
if #types == 0 then return nil end
-- rotate: start from the type AFTER the last one we picked (wrap-around)
local start = (seqCursor % #types) + 1
for step = 0, #types - 1 do
local idx = ((start - 1 + step) % #types) + 1
local h = nearestInstanceOf(types[idx])
if h then
seqCursor = idx  -- next pick advances to the following type
return h
end
end
return nil
end
local herbs = selectedHerbsSorted()
return herbs and herbs[1]
end
------------------------------------------------------------------ automation
local function tryHarvest(h, walkOnly, manual)
if log then log(("-> %s [%s]%s"):format(h.id, h.region, walkOnly and " (walk)" or ""), "info") end
if not Move.approach(h, walkOnly, manual) then
return false
end
for tries = 1, CONFIG.MaxRetries + 1 do
if not State.running and not manual then return false end
-- check if already collected by a previous finish_harvest event
if Harvest.collected[h.instanceId] then
if not Harvest.counted[h.instanceId] then
Harvest.counted[h.instanceId] = true
State.stats.harvested = State.stats.harvested + 1
end
if log then log("   already collected (ok)", "good") end
task.wait(CONFIG.HarvestCooldown)
return true
end
-- check if model still exists (already collected or despawned)
if not (h.model and h.model.Parent) then
-- if we got a result, it was collected — count as ok
local alreadyDone = Harvest.lastResult[h.instanceId]
if alreadyDone and alreadyDone.ok == true then
if not Harvest.counted[h.instanceId] then
Harvest.counted[h.instanceId] = true
State.stats.harvested = State.stats.harvested + 1
end
return true
end
-- herb gone but no result — silently skip (was likely collected by finish_harvest)
return false
end
-- walk closer if needed (pos() is nil-safe now)
local p2 = h.pos()
if not withinRange(h) and p2 then walkTo(p2); groundSnap(p2, p2.Y); task.wait(0.3) end
-- NEVER fire blind: if still out of range, re-walk next retry instead of
-- firing a remote the server rejects with "too far" (walk back-and-forth bug)
if not withinRange(h) then
if log then log("   still out of range — re-approaching", "info") end
task.wait(0.25)
else
Harvest.lastResult[h.instanceId] = nil
Harvest.collected[h.instanceId] = nil
State.invalid = false
local ok, how = firePair(h)
local res = Harvest.lastResult[h.instanceId]
-- USER RULE: finish_harvest fired = this herb is DONE. Count 1 success and
-- advance to the next herb IMMEDIATELY — never wait for a server ack that may
-- not arrive (that wait is what wedged Sequence on the first herb). Only an
-- explicit harvest_result ok=false within the wait window counts as a failure.
if ok and not (res and res.ok == false) then
Harvest.collected[h.instanceId] = true
if not Harvest.counted[h.instanceId] then
Harvest.counted[h.instanceId] = true
State.stats.harvested = State.stats.harvested + 1
end
State.invalid = false
if toast then toast(("+1 %s"):format(h.id), "good") end
if log   then log("   ok (finish_harvest fired)", "good") end
task.wait(CONFIG.HarvestCooldown)
return true
end
-- harvest result: check why
if res and res.ok == false then
local m = tostring(res.msg)
if log then log(("   server: %s"):format(m), "warn") end
-- mastery lock = can't harvest this type
if m:lower():find("mastery", 1, true) then
Harvest.failAt[h.instanceId] = os.clock()
return false
end
-- invalid movement = anti-cheat, just re-fire (don't count as fail)
if isInvalidMovement(res) then
if log then log("   invalid movement, retrying...", "info") end
State.invalid = false  -- clear, just retry
task.wait(0.5)
-- don't increment failed, just loop again
-- re-walk to position if needed
local p3 = h.pos()
if not withinRange(h) and p3 then
walkTo(p3); groundSnap(p3, p3.Y); task.wait(0.3)
end
else
-- other server error = retry
end
else
-- no result = timeout, retry
end
task.wait(0.25)
end
end
Harvest.failAt[h.instanceId] = os.clock()
return false
end
-- automation-specific connections (killed instantly on stop)
local _autoBag = {}
local autoGen = 0
local function trackAuto(c) if c then _autoBag[#_autoBag+1] = c end; return c end
local function killAuto()
State.running = false
autoGen = autoGen + 1
for _, t in ipairs(_autoBag) do pcall(task.cancel, t) end
_autoBag = {}
end
local function startAuto()
if State.running then
-- already running (e.g. boot auto-resume) — say so instead of silent no-op
if toast then toast("automation already running", "info") end
return
end
-- require at least one SELECTED type (not necessarily live — respawns resume);
-- selectedHerbsSorted()==0 would wrongly refuse to start while the world is
-- fully harvested, halting the "wait for respawn" behavior
local anySel = false
for _ in pairs(State.selectedTypes) do anySel = true; break end
if not anySel then
if toast then toast("select a herb type first", "warn") end
if log   then log("no herb types selected", "warn") end
return
end
local gen = autoGen + 1; autoGen = gen
State.running = true
persistAutoStart(true)
saveSettings()
if toast then toast("automation started", "good") end
if log   then log("automation START  mode=" .. State.mode .. "  strategy=" .. State.strategy, "good") end
trackAuto(task.spawn(function()
while alive and State.running and gen == autoGen do
-- pcall the whole iteration so a single bad herb record (destroyed model,
-- nil pos, remote error) can NEVER kill the loop silently and wedge
-- State.running = true — the "Start auto does nothing" bug
local okIt, itErr = pcall(function()
local target = pickHarvestTarget()
local progressed = false
if target then
if target and target.model and target.model.Parent then
if State.invalid then renew() end
local result = tryHarvest(target, false)
progressed = true
-- only re-sort after current herb is done (collected or failed)
end
end
if not progressed then
-- no live herb found this iteration
local anySel = false
for _ in pairs(State.selectedTypes) do anySel = true; break end
if not anySel then
-- nothing selected at all: stop cleanly instead of spinning forever
if toast then toast("no herb types selected — auto stopped", "warn") end
if log then log("auto STOP — no types selected", "warn") end
persistAutoStart(false)
saveSettings()
return "stop"
end
-- types are selected but all herbs are harvested: wait for respawns, don't halt
end
task.wait(progressed and 0.15 or CONFIG.IdleWait)
end)
if not okIt then
-- log the error and keep the loop alive (respawn may fix the record)
if log then log("auto iteration error: " .. tostring(itErr), "bad") end
task.wait(CONFIG.IdleWait)
elseif itErr == "stop" then
break
end
end
if gen == autoGen then
State.running = false
if updateStatus then updateStatus() end
if toast then toast("automation stopped", "info") end
if log   then log("automation STOP", "info") end
end
end))
end
local function stopAuto()
autoGen = autoGen + 1
killAuto()
persistAutoStart(false)
saveSettings()
end

corner(b, o.radius or R1)
local st = stroke(b, o.disabled and J.border or k.stk, 1, o.disabled and 0.6 or (k.line and 0.4 or 0.35))

local base = b.Size
local hovered = false

end)
end

end)

local function switchTab(tabName)

if emptyState then emptyState.Visible = (#knownSorted() == 0) end
end

for name, page in pairs(tabPages) do

if page.sidebarFrame then page.sidebarFrame.Visible = (name == tabName) end
end
	-- update tab button styles (idle = Text-Muted per UX_SPEC §5; hover handled by buildTabs)
	for name, btn in pairs(tabButtons) do
	local isActive = (name == tabName)
	pcall(function() btn.TextColor3 = isActive and J.text or J.muted end)
	end
-- content crossfade: fade-in the newly shown page (0.12s Quad Out, fade only)
local shown = tabPages[tabName]
if shown then
pcall(function()
local cf = shown.contentFrame

end
end)
end
	-- tab indicator slide: 2px jade bar tweens Position.X + Size.X to the active tab
	-- (0.18s Cubic Out per UX_SPEC §4); first switch snaps, later switches tween
	local btn = tabButtons[tabName]

	if btn and ind then
	local tx = btn.Position.X.Offset

	if not _indicatorInit then
	_indicatorInit = true

	else

	{Position = UDim2.new(0, tx, 1, -2), Size = UDim2.new(0, tw, 0, 2)}):Play()
	end)
	end
	end
	end

}
keyName = function(k)
if not k then return "—" end
return KEY_SHORT[k.Name] or k.Name
end
capturing = nil
stopCapture = function(cancel)
if not capturing then return end
if capturing.timer then pcall(task.cancel, capturing.timer); capturing.timer = nil end
if capturing.pulse then pcall(function() capturing.pulse:Cancel() end); capturing.pulse = nil end
if not cancel and capturing.button then capturing.button.Text = keyName(Binds[capturing.action]) end
if capturing.button then
pcall(function()
local st = capturing.button.UIStroke
if st then st.Color = J.border; st.Thickness = 1; st.Transparency = 0.3 end
end)
pcall(function() capturing.button.TextColor3 = J.text end)
end
capturing = nil
end
startCapture = function(action, button)
stopCapture(true)
capturing = {action = action, button = button}
button.Text = "PRESS KEY…"; button.TextColor3 = J.gold
pcall(function()
local st = button.UIStroke
if st then st.Color = J.gold; st.Thickness = 2; st.Transparency = 0 end
end)
-- 1.2s gold pulse loop on the stroke (TweenService)
local st = button.UIStroke
if st then
local tw = TweenSvc:Create(st, TweenInfo.new(1.2, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, 0, -1), {Transparency = 0.7})
capturing.pulse = tw
pcall(function() tw:Play() end)
end
capturing.timer = task.delay(6, function() stopCapture(true) end)
end
------------------------------------------------------------------ log (pooled, 200-row circular buffer)
local LOG_MAX = 200
local logPool = {}
local logRowCount = 0
local LOG_COL = {
good = J.ok, warn = J.textGold, bad = J.dangerText, info = J.textSecondary,
}
local LOG_TICK = {
good = J.ok, warn = J.gold, bad = J.danger, info = J.seriesIce,
}
guiLog = function(msg, kind)
kind = kind or "info"
if not gui.logFrame then print("[Jianghu] " .. tostring(msg)); return end
local col = LOG_COL[kind] or J.textSecondary
local tickc = LOG_TICK[kind] or J.seriesIce
local row
-- reuse from pool or create new
if #logPool > 0 then
row = table.remove(logPool)
else
row = mk("Frame", {Size = UDim2.new(1,0,0,22), BackgroundTransparency = 1}, gui.logFrame)
end
row.Parent = gui.logFrame
row.BackgroundTransparency = 1
-- update existing children instead of creating new
local tick = row:FindFirstChild("tick")
local ts = row:FindFirstChild("ts")
local body = row:FindFirstChild("body")
if not tick then
tick = mk("Frame", {Name="tick", Size = UDim2.new(0,3,0,3), Position = UDim2.new(0,0,0,9.5), BorderSizePixel = 0}, row); corner(tick, 2)
ts = mlabel({Name="ts", TextSize = 10, TextColor3 = J.muted,
Position = UDim2.new(0,12,0,0), Size = UDim2.new(0,56,1,0)}, row)
body = label({Name="body", TextSize = 12, TextColor3 = J.textSecondary,
Position = UDim2.new(0,76,0,0), Size = UDim2.new(1,-80,1,0)}, row)
end
tick.BackgroundColor3 = tickc
tick.BackgroundTransparency = 1
ts.Text = os.date("%H:%M:%S")
ts.TextTransparency = 1
body.Text = tostring(msg)
body.TextColor3 = col
body.TextTransparency = 1
logRowCount = logRowCount + 1
pcall(function() TweenSvc:Create(tick, TweenInfo.new(0.25), {BackgroundTransparency = 0}):Play() end)
pcall(function() TweenSvc:Create(ts, TweenInfo.new(0.25), {TextTransparency = 0}):Play() end)
pcall(function() TweenSvc:Create(body, TweenInfo.new(0.25), {TextTransparency = 0}):Play() end)
if logRowCount > LOG_MAX then
-- recycle oldest: move to pool instead of destroying
local oldest
for _, k in ipairs(gui.logFrame:GetChildren()) do
if k:IsA("Frame") then oldest = k break end
end
if oldest and oldest:IsA("Frame") then
oldest.Parent = nil
logRowCount = logRowCount - 1
logPool[#logPool + 1] = oldest
end
end
gui.logFrame.CanvasPosition = Vector2.new(0, gui.logLayout.AbsoluteContentSize.Y)
end
local function clearLog()
if not gui.logFrame then return end
for _, k in ipairs(gui.logFrame:GetChildren()) do if k:IsA("Frame") then k:Destroy() end end
logRowCount = 0
logPool = {}
end

reconcileList = function()
if not listFrame then return end
-- prune stale harvest marks: instanceIds that no longer exist in the live
-- scan (harvested + despawned) can never be re-picked, so drop their marks
-- to keep the collected/counted tables bounded across respawns
for k in pairs(Harvest.collected) do if not State.byId[k] then Harvest.collected[k] = nil end end
for k in pairs(Harvest.counted)   do if not State.byId[k] then Harvest.counted[k] = nil end end
for k in pairs(Harvest.lastResult) do if not State.byId[k] then Harvest.lastResult[k] = nil end end
for k in pairs(Harvest.failAt) do
if not State.byId[k] then Harvest.failAt[k] = nil
elseif Harvest.failAt[k] and os.clock() - Harvest.failAt[k] > 30 then Harvest.failAt[k] = nil
end
end
local seen = {}
local sorted = knownSorted()
for i, id in ipairs(sorted) do
seen[id] = true
if not rows[id] then rows[id] = herbCard(id) end
rows[id].frame.LayoutOrder = i
rows[id].refreshLocked()
-- always repaint to sync visual with selection state
rows[id].paintSel()
end
-- destroy cards for herb types that vanished from the scan (keeps rows bounded)
for id, r in pairs(rows) do
if not seen[id] then
pcall(function() r.frame:Destroy() end)
rows[id] = nil
end
end
setEmptyState(#sorted == 0)
guiUpdateStatus()
end

local function selectNearestTypes(n)
local t = {}
for id in pairs(State.byType) do
if not isTypeLocked(id) then
local h = nearestInstanceOf(id)
t[#t+1] = {id = id, d = h and distTo(h) or 1e9}
end
end
table.sort(t, function(a, b) return a.d < b.d end)
State.selectedTypes = {}
for i = 1, math.min(n, #t) do State.selectedTypes[t[i].id] = true end
if onHerbsChanged then onHerbsChanged() end
if toast then toast("selected nearest types", "info") end
saveSettings()
end
------------------------------------------------------------------ window visibility / minimize
local _hidePending = false
local _entered = false  -- first show runs the content-zone entrance stagger (UX_SPEC §4)
local setMinimized = nil  -- forward decl (defined below; called from setVisible restore-on-show)
setVisible = function(v)
if not main.frame then return end
if v then
-- restored on show: a minimized window comes back with its full layout
if minimized then setMinimized(false) end
-- show: fade + scale 0.96->1 (0.3s Cubic Out)
_hidePending = false
main.frame.Visible = true
main.frame.GroupTransparency = 1
local sc = main.frame:FindFirstChildOfClass("UIScale")
if sc then pcall(function() sc.Scale = 0.96 end) end
TweenSvc:Create(main.frame, TweenInfo.new(0.3, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out), {GroupTransparency = 0}):Play()
if sc then pcall(function() TweenSvc:Create(sc, TweenInfo.new(0.3, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out), {Scale = 1}):Play() end) end
-- content zones stagger: header/tabs/content/status fade in +0.04s apart (first show only, max 0.4s)
if not _entered then
_entered = true
local zones = {headerFrame, tabsBar, content.frame, statusBar}
for zi = 1, #zones do
local zone = zones[zi]
if zone then
zone.GroupTransparency = 1
task.delay(0.04 * zi, function()
pcall(function()
TweenSvc:Create(zone, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {GroupTransparency = 0}):Play()
end)
end)
end
end
end
else
-- hide: fade 0.18s Quad In, then hide on completion (guarded against toggle races)
local tw = TweenSvc:Create(main.frame, TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {GroupTransparency = 1})
_hidePending = true
tw:Play()
tw.Completed:Connect(function()
if _hidePending and main.frame then main.frame.Visible = false end
_hidePending = false
end)
end
end
setMinimized = function(v)
minimized = v
if not main.frame then return end
local rh = main.frame:FindFirstChild("fkResizeHandle")
if v then
-- minimized = header only: keep the 64px header visible, hide tabs + content, drop the resize grip
if content.frame then content.frame.Visible = false end
if tabsBar then tabsBar.Visible = false end

function refreshSettingsHighlight()
for group, info in pairs(settingsSegs) do
local cur = info.getter and info.getter() or nil
local n = info.n or #info.values
local idx = 1
for i, val in ipairs(info.values) do if val == cur then idx = i; break end end
for i, b in ipairs(info.labels) do
TweenSvc:Create(b, TweenInfo.new(0.16), {TextColor3 = (i == idx) and J.ink or J.silver}):Play()

{Position = UDim2.new((idx-1)/n, 2, 0, 3)}):Play()

TweenSvc:Create(trackB, TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
{BackgroundColor3 = disabled and J.raised or (on and J.jade or J.sunken)}):Play()
local g = trackB:FindFirstChild("tg")
if on and not disabled then
if not g then g = grad(trackB, J.jade, J.jadeDk, 90); if g then g.Name = "tg" end end
else
if g then g:Destroy() end
end
-- thumb fill: paper when enabled, muted when disabled
TweenSvc:Create(knob, TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
{BackgroundColor3 = disabled and J.muted or J.paper}):Play()
end

set(not get()); paint(); saveSettings()
end)

end

function openSettings()
-- settings sits inside content.frame: a minimized window must restore its layout first
if minimized then setMinimized(false) end
if not settingsFrame then buildSettings() end
preSettingsTab = activeTab  -- remember which tab we were on
-- reset panel state while hidden (no flash on reveal)
if gui.settingsPanel then
gui.settingsPanel.Position = UDim2.new(0.5,0,0.5,0)
gui.settingsPanel.BackgroundTransparency = 0
gui.settingsPanel.GroupTransparency = 1
local us = gui.settingsPanel:FindFirstChildOfClass("UIScale")
if us then us.Scale = 0.98 end
end
-- hide main content, show settings
if harvestSidebar then harvestSidebar.Visible = false end
if harvestContent then harvestContent.Visible = false end
-- hide empty state too
if emptyState then emptyState.Visible = false end
if listFrame then listFrame.Visible = false end
settingsFrame.Visible = true
if gui.settingsScroll then gui.settingsScroll.CanvasPosition = Vector2.new(0,0) end
-- scrim fade 0.18s Linear (settings scrim 0.65)
TweenSvc:Create(settingsFrame, TweenInfo.new(0.18, Enum.EasingStyle.Linear), {BackgroundTransparency = 0.35}):Play()
-- panel: scale 0.98 -> 1 + fade, 0.22s Cubic Out
if gui.settingsPanel then
TweenSvc:Create(gui.settingsPanel, TweenInfo.new(0.22, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out), {GroupTransparency = 0}):Play()
local us = gui.settingsPanel:FindFirstChildOfClass("UIScale")
if us then TweenSvc:Create(us, TweenInfo.new(0.22, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out), {Scale = 1}):Play() end
end
refreshSettingsHighlight()
end

-- hide any lingering body-auto-bt confirm overlay (gear button isn't covered by it)
if _bodyConfirmOverlay and _bodyConfirmOverlay.Parent then
_bodyConfirmOverlay.Visible = false
end
if gui.settingsPanel then
-- fade scrim (0.18s Linear) + panel out (0.18s Quad In), restore on completion
local tw = TweenSvc:Create(settingsFrame, TweenInfo.new(0.18, Enum.EasingStyle.Linear), {BackgroundTransparency = 1})
TweenSvc:Create(gui.settingsPanel, TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {GroupTransparency = 1}):Play()

switchTab(preSettingsTab)

if harvestContent then harvestContent.Visible = true end

-- ─── AUTO-BREAKTHROUGH ENGINE (IIFE body level — accessible everywhere) ───
local function pct(x) local n = x or 0; if n ~= n then return "--" end return ("%.1f%%"):format(n * 100) end
esc = function(t)
local s = tostring(t)
s = s:gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;")
return s
end
doBreakthrough = function(path)
local folder = Replicated:FindFirstChild("Network")
if not folder then autoBt.feedback = "no network"; return end
local cultFolder = folder:FindFirstChild("Cultivation")
if not cultFolder then autoBt.feedback = "no cult folder"; return end
local remote = cultFolder:FindFirstChild("breakthrough")
if not (remote and remote:IsA("RemoteEvent")) then autoBt.feedback = "no breakthrough remote"; return end
pcall(function() remote:FireServer(path) end)
autoBt.lastFire = tick()
autoBt.awaiting = true
autoBt.awaitPath = path
autoBt.awaitStage = (cultState and cultState[path] and cultState[path].Stage) or "?"
autoBt.flashUntil = tick() + 1.2
local chNow = (cultState and cultState[path] and cultState[path].BreakthroughChance) or 0
autoBt.feedback = "fired " .. esc(path) .. " @ " .. pct(chNow) .. " · cooldown " .. AUTOBT_COOLDOWN .. "s"
end

local function romanVal(t)
local map = {I=1,V=5,X=10,L=50,C=100,D=500,M=1000}
local total, prev = 0, 0
for i = #t, 1, -1 do
local v = map[t:sub(i,i)] or 0
total = total + (v < prev and -v or v); prev = v
end
return total
end
local function parseStage(stage)
if type(stage) ~= "string" then return "", nil end
local prefix, tail = stage:match("^(.-)%s+([IVXLCDM]+)$")
if not prefix then return stage, nil end
return prefix, romanVal(tail)
end
local function normPath(p)
if type(p) ~= "string" then return nil end
local t = p:match("^%s*(.-)%s*$") or p
if t == "" or t:lower() == "none" then return nil end
return t
end
local function addCommas(n)
if type(n) == "number" and math.abs(n) >= 1e15 then return tostring(n) end
local s = string.format("%d", math.floor(n + 0.5))
local rev = s:reverse():gsub("(%d%d%d)", "%1,"):gsub(",$", "")
return rev:reverse()
end
-- strip trailing zeros (and the dot) from a fixed-2-decimal string: "50.00"→"50", "75.60"→"75.6", "75.67"→"75.67"
local function trimDec(s) return (s:gsub("%.?0+$", "")) end
local function fmtN(n)
if type(n) ~= "number" or n ~= n then return "—" end
local a = math.abs(n)
if a >= 1e9 then return trimDec(("%.2f"):format(n/1e9)) .. "B (" .. addCommas(n) .. ")" end
if a >= 1e6 then return trimDec(("%.2f"):format(n/1e6)) .. "M (" .. addCommas(n) .. ")" end
if a >= 1e3 then return trimDec(("%.2f"):format(n/1e3)) .. "K (" .. addCommas(n) .. ")" end
return addCommas(n)
end
local function fmtRate(n)
if type(n) ~= "number" or n ~= n then return "—" end
return fmtN(n) .. "/s"
end
local function fmtT(s)
if s == nil or s ~= s or math.abs(s) > 1e15 then return "—" end
if s <= 0 then return "0s" end
local d  = math.floor(s / 86400)
local h  = math.floor((s % 86400) / 3600)
local m  = math.floor((s % 3600) / 60)
local sc = math.floor(s % 60)
if d > 0 then return string.format("%dd %dh %dm", d, h, m) end
if h > 0 then return string.format("%dh %dm %ds", h, m, sc) end
if m > 0 then return string.format("%dm %ds", m, sc) end
return string.format("%ds", sc)
end

-- chance model (quadratic concave fit)
local function solve3(m, t)
local M = {
{m[1][1],m[1][2],m[1][3],t[1]},
{m[2][1],m[2][2],m[2][3],t[2]},
{m[3][1],m[3][2],m[3][3],t[3]},
}
for ccol = 1, 3 do
local piv, best = ccol, math.abs(M[ccol][ccol])
for r = ccol+1, 3 do
local a = math.abs(M[r][ccol])
if a > best then best = a; piv = r end
end
if best < 1e-12 then return nil end
if piv ~= ccol then M[ccol], M[piv] = M[piv], M[ccol] end
local d = M[ccol][ccol]
for r = 1, 3 do
if r ~= ccol then
local f = M[r][ccol] / d
for c2 = ccol, 4 do M[r][c2] = M[r][c2] - f * M[ccol][c2] end
end
end
end
return { M[1][4]/M[1][1], M[2][4]/M[2][2], M[3][4]/M[3][3] }
end
local function fitChance()
local n = #cultSamples
if n < 2 then return nil end
local S0,S1,S2,S3,S4 = 0,0,0,0,0
local T0,T1,T2 = 0,0,0
for i = 1, n do
local r, ch = cultSamples[i].r, cultSamples[i].ch
local r2 = r*r; local r3 = r2*r; local r4 = r3*r
S0 = S0+1; S1 = S1+r; S2 = S2+r2; S3 = S3+r3; S4 = S4+r4
T0 = T0+ch; T1 = T1+r*ch; T2 = T2+r2*ch
end
local lin = nil
local det2 = S0*S2 - S1*S1
if math.abs(det2) > 1e-12 then
lin = { (T0*S2 - T1*S1)/det2, (S0*T1 - S1*T0)/det2 }
end
local quad = nil
if n >= 6 then
quad = solve3({{S0,S1,S2},{S1,S2,S3},{S2,S3,S4}}, {T0,T1,T2})
end
return { n = n, lin = lin, quad = quad }
end
local function useQuad(fit)
if not fit then return false end
return (fit.quad ~= nil and fit.n >= 10 and math.abs(fit.quad[3]) > 1e-7)
end
local function chanceAtCost(fit)
if not fit then return nil, "none" end
if useQuad(fit) then
local a,b,c = fit.quad[1], fit.quad[2], fit.quad[3]
return math.min(1, math.max(0, a+b+c)), "quad"
elseif fit.lin then
local a,b = fit.lin[1], fit.lin[2]
return math.min(1, math.max(0, a+b)), "linear"
end
return nil, "none"
end

local function vertexInfo(fit)
    if not (fit and fit.quad) then return nil end
    local a,b,c = fit.quad[1], fit.quad[2], fit.quad[3]
    if math.abs(c) <= 1e-7 then return nil end
    local rv = -b/(2*c)
    return { r = rv, ch = math.min(1, math.max(0, a+b*rv+c*rv*rv)), concave = c < 0 }
end
local function attemptTimeFor(T, fit, curR, cur, cost, rate, chance)
    if not rate or rate <= 0 then return nil, "rate0" end
    if chance and chance >= T then
        if cost and cur >= cost then return 0, "attempt now" end
        local t = (cost and cost > 0) and ((cost - cur) / rate) or nil
        return t, "chance met now; gate = essence"
    end
    local function rForTarget(fit, T, curR)
        if not fit then return nil, nil end
        local v = vertexInfo(fit)
        if useQuad(fit) then
            local a,b,c = fit.quad[1], fit.quad[2], fit.quad[3]
            if math.abs(c) < 1e-12 then
                if math.abs(b) < 1e-12 then return nil, v end
                return (T-a)/b, v
            end
            local disc = b*b - 4*c*(a - T)
            if disc < 0 then return nil, v end
            local sq = math.sqrt(disc)
            local r1 = (-b - sq)/(2*c)
            local r2 = (-b + sq)/(2*c)
            local cand = {}
            if r1 >= curR - 1e-9 then table.insert(cand, r1) end
            if r2 >= curR - 1e-9 then table.insert(cand, r2) end
            if #cand == 0 then return nil, v end
            table.sort(cand)
            return cand[1], v
        elseif fit.lin then
            local a,b = fit.lin[1], fit.lin[2]
            if math.abs(b) < 1e-12 then return nil, v end
            return (T-a)/b, v
        end
        return nil, v
    end
    local rT, v = rForTarget(fit, T, curR)
    if not rT then
        if v and v.concave and v.r and v.r >= curR - 1e-9 and v.ch and v.ch < T then
            return nil, string.format("peaks %.0f%%", v.ch*100)
        end
        return nil, "unreachable"
    end
    local rUse = math.max(rT, 1)
    local need = rUse*cost - cur
    if need < 0 then need = 0 end
    local t = need / rate
    local tag = (rT <= 1.0001) and "ok@cost" or "overfill"
    return t, tag
end

local function _skeleton(card, on)
if not card then return end
local skel = card:FindFirstChild("skeleton")
if not skel then return end
if on then
skel.Visible = true
if not _skeletonTweens[card] then
_skeletonTweens[card] = TweenSvc:Create(skel, TweenInfo.new(1.6, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true), {BackgroundTransparency = 0.8})
pcall(function() _skeletonTweens[card]:Play() end)
end
else
skel.Visible = false
local tw = _skeletonTweens[card]
if tw then pcall(function() tw:Cancel() end); _skeletonTweens[card] = nil end
end
end
-- count-up 0.12s; snap when |Δ| >= 25% of target (UX_SPEC §4 anti-flicker)
local function countUp(lbl, target, fmt)
local cur = tonumber(lbl:GetAttribute("cv")) or target
if cur ~= cur then cur = 0 end
if target ~= target then target = 0 end
fmt = fmt or fmtN
lbl:SetAttribute("cv", target)
if math.abs(target - cur) >= math.max(1, math.abs(target) * 0.25) then
pcall(function() lbl.Text = fmt(target) end)
return
end
local t0 = os.clock()
task.spawn(function()
while alive and lbl and lbl.Parent do
local f = (os.clock() - t0) / 0.12
if f >= 1 then
pcall(function() lbl.Text = fmt(target) end)
return
end
pcall(function() lbl.Text = fmt(cur + (target - cur) * (f * (2 - f))) end)
task.wait(1 / 30)
end
end)
end

-- ─── DASHBOARD TICKS (4Hz; direct writes only — UX_SPEC §4 anti-flicker) ───
local function tickSession()
local now = os.clock()
local harvested = State.stats.harvested or 0
if State.running and not _sess.prevRunning then
_sess.start = now; _sess.h0 = harvested; _sess.last = harvested
_sess.perMin = {}; for bi = 1, 12 do _sess.perMin[bi] = 0 end
_sess.acc = 0; _sess.lastMin = now; _sess.max = 1
if _dash.c6.deltaTw then pcall(function() _dash.c6.deltaTw:Cancel() end); _dash.c6.deltaTw = nil end
_dash.c6.delta.TextTransparency = 1
end
_sess.prevRunning = State.running
local el = now - _sess.start
local rate = (el > 0.5) and ((harvested - _sess.h0) / el) or 0
_dash.c6.herbs.Text = fmtN(harvested)
_dash.c6.rate.Text = fmtRate(rate)
if el >= 3600 then
_dash.c6.time.Text = string.format("%d:%02d:%02d", math.floor(el / 3600), math.floor((el % 3600) / 60), math.floor(el % 60))
else
_dash.c6.time.Text = string.format("%02d:%02d", math.floor(el / 60), math.floor(el % 60))
end
local d = harvested - _sess.last
if d > 0 then
_sess.last = harvested
_sess.acc = _sess.acc + d
if _dash.c6.deltaTw then pcall(function() _dash.c6.deltaTw:Cancel() end); _dash.c6.deltaTw = nil end

local function tickTrend()
local n = #cultSamples
local w = well.AbsoluteSize.X - 16
local h = 80
if n < 2 then
_dash.c3.pill.Text = ""
_dash.c3.fitTag.Text = "need more samples"
_dash.r2 = nil
if _dash.c3.count and _dash.c3.count ~= 0 then
_dash.c3.count = 0
for di = 1, 60 do _dash.c3.dots[di].Visible = false end
for si = 1, 59 do _dash.c3.segs[si].Visible = false end
_dash.c3.vtx.Visible = false
end
return
end
local minCh, maxCh = 1, 0
for i = 1, n do
local ch = cultSamples[i].ch
if ch < minCh then minCh = ch end
if ch > maxCh then maxCh = ch end
end
local pad = (maxCh - minCh) * 0.1
local lo, hi = minCh - pad, maxCh + pad
if hi - lo < 1e-9 then hi = lo + 0.05 end
if _dash.c3.count ~= n then
_dash.c3.count = n
for di = 1, 60 do _dash.c3.dots[di].Visible = (di <= n) end
for si = 1, 59 do _dash.c3.segs[si].Visible = (si < n) end
end
for di = 1, math.min(60, n) do
local s = cultSamples[di]
_dash.c3.dots[di].Position = UDim2.new(0, 8 + (di - 1) / 59 * w, 0, 8 + (1 - (s.ch - lo) / (hi - lo)) * h)
end
local fit = fitChance()
local a, b, c = 0, 0, 0
local isQuad = useQuad(fit)
if isQuad then a, b, c = fit.quad[1], fit.quad[2], fit.quad[3]
elseif fit and fit.lin then a, b = fit.lin[1], fit.lin[2] end
local mean = 0
for i = 1, n do mean = mean + cultSamples[i].ch end
mean = mean / n
local ssRes, ssTot = 0, 0
local px, py = {}, {}
local rMin, rMax = cultSamples[1].r, cultSamples[1].r
for i = 1, n do
local s = cultSamples[i]
if s.r < rMin then rMin = s.r end
if s.r > rMax then rMax = s.r end
end
for i = 1, n do
local s = cultSamples[i]
local mv
if isQuad then mv = a + b * s.r + c * s.r * s.r
elseif fit and fit.lin then mv = a + b * s.r
else mv = nil end
if mv ~= nil then mv = math.min(1, math.max(0, mv)) end
local diff = s.ch - (mv or s.ch)
ssRes = ssRes + diff * diff
local d2 = s.ch - mean
ssTot = ssTot + d2 * d2
end
_dash.r2 = (ssTot < 1e-12) and 1 or (1 - ssRes / ssTot)
_dash.c3.fitTag.Text = string.format("R\194\178 %.3f \194\183 %s fit", _dash.r2, isQuad and "quad" or "linear")
_dash.c3.pill.Text = "P" .. n .. " \194\183 " .. fmtN((cultState[cultActive] or {}).Current)
if isQuad or (fit and fit.lin) then
local step = (rMax - rMin) / 59
if step < 1e-12 then step = 1 end
for i = 1, 60 do
local r = rMin + (i - 1) * step
local mv
if isQuad then mv = math.min(1, math.max(0, a + b * r + c * r * r))
else mv = math.min(1, math.max(0, a + b * r)) end
px[i] = 8 + (i - 1) / 59 * w
py[i] = 8 + (1 - (mv - lo) / (hi - lo)) * h
end
for si = 1, 59 do
local p1x, p1y, p2x, p2y = px[si], py[si], px[si + 1], py[si + 1]
local dx, dy = p2x - p1x, p2y - p1y
local L = math.max(1, math.sqrt(dx * dx + dy * dy))
local s = _dash.c3.segs[si]
s.Size = UDim2.new(0, L, 0, 1)
s.Rotation = math.atan2(dy, dx) * 180 / math.pi
TweenSvc:Create(s, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Position = UDim2.new(0, p1x + dx / 2, 0, p1y + dy / 2)}):Play()
end
end
local vtx = vertexInfo(fit)
if vtx and vtx.r then
local vx = 8 + math.clamp((vtx.r - rMin) / (rMax - rMin + 1e-9), 0, 1) * w
local vch = math.min(1, math.max(0, vtx.ch))
_dash.c3.vtx.Position = UDim2.new(0, vx, 0, 8 + (1 - (vch - lo) / (hi - lo)) * h)
_dash.c3.vtx.Visible = true
elseif _dash.c3.vtx.Visible then
_dash.c3.vtx.Visible = false
end
end
