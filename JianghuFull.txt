local function ingest(model, found, list)
if found[model] then return end
local rec = tracked[model]
if not rec then
local ok = pcall(function() rec = makeRecord(model) end)
if ok and rec then tracked[model] = rec else return end
else
pcall(function()
rec.root   = model:FindFirstChild("Root", true)
rec.prompt = model:FindFirstChild("HarvestPrompt", true) or model:FindFirstChildOfClass("ProximityPrompt")
if not rec.id then local a = model:GetAttributes(); rec.id = a[CONFIG.HerbAttr] end
end)
end
if rec and rec.id then found[model] = true; list[#list+1] = rec end
end

-- recursive walk: no GetDescendants allocation, yields cooperatively
local function walkScan(root, cap, found, list, counter)
local stack = {root}
while #stack > 0 do
local node = table.remove(stack)
counter[1] = counter[1] + 1
if counter[1] > cap then return true end
pcall(function()
if node:GetAttribute(CONFIG.HerbAttr) then ingest(node, found, list) end
end)
local ok, children = pcall(function() return node:GetChildren() end)
if ok and children then
for _, child in ipairs(children) do
stack[#stack + 1] = child
end
end
-- yield every 2000 nodes to prevent frame drops
if counter[1] % 2000 == 0 then task.wait() end
end
return false
end
local function gcTracked()
for model, rec in pairs(tracked) do
if not model.Parent then detachRecord(rec); tracked[model] = nil end
end
end
local function finalizeScan(list, nodes, verbose, capped)
local byType = {}
for _, rec in ipairs(list) do
byType[rec.id] = byType[rec.id] or {}
byType[rec.id][#byType[rec.id]+1] = rec
end
State.herbs, State.byType = list, byType
State.byId = {}
for _, rec in ipairs(list) do
State.byId[rec.instanceId] = rec
knownTypes[rec.id] = true
end
for id in pairs(State.selectedTypes) do
if not byType[id] then State.selectedTypes[id] = nil end
end
-- prune knownTypes: remove types with zero live instances
for id in pairs(knownTypes) do
if not byType[id] then knownTypes[id] = nil end
end
State.lastScanNodes  = nodes
State.lastScanHerbs  = #list
if onHerbsChanged then onHerbsChanged() end
pcall(function() if shimmer then shimmer() end end)
-- suppress scan logs during first 5s after boot to avoid startup spam
local bootTime = bootTime or os.clock()
if verbose or (#list ~= State._prevCount and (os.clock() - bootTime) > 5) then
State._prevCount = #list
if log then log("scan: " .. nodes .. " nodes" .. (capped and " (cap)" or "") .. ", " .. #list .. " herbs", #list > 0 and "good" or "info") end
end
end
local function fullScan(verbose)
setScanning(true)
gcTracked()
local found, list, counter = {}, {}, {}; counter[1] = 0
local roots = priorityRoots()
local capped = false
for _, root in ipairs(roots) do
if walkScan(root, CONFIG.NodeCap, found, list, counter) then capped = true; break end
end
if not capped then
for _, c in ipairs(Workspace:GetChildren()) do
if (c:IsA("Model") or c:IsA("Folder")) and not found[c] then
local already = false
for _, r in ipairs(roots) do if r == c then already = true; break end end
if not already then
if walkScan(c, CONFIG.NodeCap - counter[1], found, list, counter) then capped = true; break end
end
end
end
end
finalizeScan(list, counter[1], verbose, capped)
setScanning(false)
end
local function quickScan()
local found, list, counter = {}, {}, {}; counter[1] = 0
for _, root in ipairs(priorityRoots()) do
if walkScan(root, 40000, found, list, counter) then break end
end
finalizeScan(list, counter[1], false, false)
end

local function groundYAt(x, z, refY)
local originY = (refY or 0) + 2
local hit = Workspace:Raycast(Vector3.new(x, originY, z), Vector3.new(0, -40, 0), _groundParams)
if hit then
local hy = hit.Position.Y
if refY and hy < refY - 10 then return refY end
return hy
end
return refY
end
local function groundSnap(pos, refY)
local y = groundYAt(pos.X, pos.Z, refY or pos.Y); if not y then return end
local r = hrp(); if not r then return end
r.CFrame = CFrame.new(pos.X, y + CONFIG.HalfHeight, pos.Z) * (r.CFrame - r.CFrame.Position)
pcall(function() r.Velocity = Vector3.zero end)
pcall(function() r.AssemblyLinearVelocity = Vector3.zero end)
local h = hum()
if h then pcall(function() h:ChangeState(Enum.HumanoidStateType.GettingUp) end) end
end
local function distSqTo(h)
local r = hrp()
local p = h and h.pos and h.pos()
if not (r and p) then return 1e18 end
local d = r.Position - p
return d.X*d.X + d.Z*d.Z
end
local function distTo(h)
local r = hrp()
local p = h and h.pos and h.pos()
return (r and p) and (r.Position - p).Magnitude or 1e9
end
local function withinRange(h)
local maxD = 8
pcall(function() if h.prompt and h.prompt.MaxActivationDistance > 0 then maxD = h.prompt.MaxActivationDistance end end)
return distTo(h) <= (maxD - CONFIG.PromptMargin)
end
-- WALK: pure pathfinding along the ground
local _lastPathTarget, _lastPath = nil, nil
local function walkTo(target)
local r, h = hrp(), hum(); if not r or not h then return false end
local saved = h.WalkSpeed
pcall(function() h.WalkSpeed = CONFIG.WalkSpeed end)
pcall(function()
if not _lastPath or not _lastPathTarget or (target - _lastPathTarget).Magnitude > 2 then
local path = PathSvc:CreatePath({AgentRadius = 2, AgentHeight = 5, AgentCanJump = true})
path:ComputeAsync(r.Position, target)
_lastPath = path
_lastPathTarget = target
end
if _lastPath.Status == Enum.PathStatus.Success then
for _, wp in ipairs(_lastPath:GetWaypoints()) do
h:MoveTo(wp.Position)
if wp.Action == Enum.PathWaypointAction.Jump then pcall(function() h.Jump = true end) end
if not h.MoveToFinished:Wait() then break end
end
else
h:MoveTo(target); task.wait(0.6)
end
end)
pcall(function() h.WalkSpeed = saved end)
return true
end
-- TWEEN: CFrame through obstacles, underground, sky — ignores terrain
local function tweenToTarget(target)
local r = hrp(); if not r then return end
local from = r.CFrame
local to = CFrame.new(target) * (from - from.Position)
local dur = math.max(0.15, (from.Position - to.Position).Magnitude / math.max(1, CONFIG.WalkSpeed))
local tw = TweenSvc:Create(r, TweenInfo.new(dur, Enum.EasingStyle.Linear), {CFrame = to})
tw:Play(); tw.Completed:Wait()
end
-- TELEPORT: instant CFrame, land exactly on the surface the herb sits on
local function teleportToTarget(target)
local r = hrp(); if not r then return end
-- raycast from just above the herb's position (handles underground platforms)
local groundY = groundYAt(target.X, target.Z, target.Y + 4)
if not groundY then groundY = target.Y end
-- teleport to ground + half height (feet exactly on surface)
r.CFrame = CFrame.new(target.X, groundY + CONFIG.HalfHeight, target.Z) * (r.CFrame - r.CFrame.Position)
pcall(function() r.Velocity = Vector3.zero end)
pcall(function() r.AssemblyLinearVelocity = Vector3.zero end)
task.wait(0.05)
local h = hum()
if h then
pcall(function() h:ChangeState(Enum.HumanoidStateType.GettingUp) end)
pcall(function() h:ChangeState(Enum.HumanoidStateType.Running) end)
end
end
-- MODE DISPATCH: each mode has its own approach logic
function Move.approach(h, walkOnly)
local target = h.pos(); local r = hrp()
if not target or not r then return false end
if not State.running then return false end
local mode = walkOnly and "Walk" or State.mode
if mode == "Walk" then
-- pure pathfinding walk along the ground
walkTo(target); groundSnap(target, target.Y); task.wait(CONFIG.Dwell)
if not withinRange(h) then walkTo(target); groundSnap(target, target.Y); task.wait(0.4) end
return withinRange(h)
end
if mode == "Tween" then
-- tween directly to herb through obstacles, then ground snap before harvest
tweenToTarget(target); task.wait(CONFIG.PostHopSettle)
-- stand on ground at herb position before firing
groundSnap(target, target.Y); task.wait(CONFIG.Dwell)
if not withinRange(h) then
walkTo(target); groundSnap(target, target.Y); task.wait(0.4)
end
return withinRange(h)
end
if mode == "Teleport" then
-- teleport directly to herb, ground snap ensures feet on surface
teleportToTarget(target); task.wait(CONFIG.Dwell)
if not withinRange(h) then
-- if still not in range, walk the last bit
walkTo(target); groundSnap(target, target.Y); task.wait(0.4)
end
return withinRange(h)
end
if mode == "Adaptive" then
local d = distTo(h)
if d > 30 then
-- far: teleport NEAR herb on same surface, then walk the rest
local dir = (target - r.Position).Unit
local nearPoint = target - dir * 12
-- use herb's Y directly (it's on the same platform)
nearPoint = Vector3.new(nearPoint.X, target.Y, nearPoint.Z)
teleportToTarget(nearPoint); task.wait(CONFIG.PostHopSettle)
end
-- walk to herb from current position
walkTo(target); groundSnap(target, target.Y); task.wait(CONFIG.Dwell)
if not withinRange(h) then walkTo(target); groundSnap(target, target.Y); task.wait(0.4) end
return withinRange(h)
end
return false
end

renewKeywords = {"too far","invalid token","renew","expired","token","bound","out of range","not in range"},
}
local function attachHarvestListeners()
pcall(function()
local folder = Replicated:FindFirstChild("Network"); if not folder then return end
for _, o in ipairs(folder:GetDescendants()) do
if o:IsA("RemoteEvent") then
track(o.OnClientEvent:Connect(function(...)
local a = tpack(...)
if not alive then return end
local id = a[1]
-- harvest_result: server tells us the outcome
if o.Name == "harvest_result" then
local ok, msg = a[2], a[3]
if type(id) == "string" then Harvest.lastResult[id] = {ok = ok, msg = msg} end
if ok == false and type(msg) == "string" then
local low = msg:lower()
-- only flag invalid for non-collection errors
if not Harvest.collected[id] then
for _, kw in ipairs(Harvest.renewKeywords) do
if low:find(kw, 1, true) then State.invalid = true; break end
end
end
end
end
-- harvest_started: server accepted our start_harvest, we're in progress
if o.Name == "harvest_started" then
if type(id) == "string" then
-- mark as in-progress (prevents "herb gone" false fail)
Harvest.lastResult[id] = Harvest.lastResult[id] or {ok = nil, msg = "started"}
end
end
-- finish_harvest from server = herb collected, regardless of amount
if o.Name == "finish_harvest" and type(id) == "string" then
Harvest.collected[id] = true
Harvest.lastResult[id] = {ok = true, msg = tostring(a[2] or "collected")}
end
end))
end
end
end)
end
local function viewportArgs(root)
local c = cam()
if not c or not root then return 0.5, 0.5, 20 end
local depth = (c.CFrame.Position - root.Position).Magnitude
if depth < 1 then depth = 1 end
local nx, ny = 0.5, 0.5
pcall(function()
local vp = c:WorldToViewportPoint(root.Position)
local sz = c.ViewportSize
nx = (sz.X > 0 and vp.X / sz.X or 0.5)
ny = (sz.Y > 0 and vp.Y / sz.Y or 0.5)
if not (nx >= 0.03 and nx <= 0.97 and ny >= 0.03 and ny <= 0.97 and vp.Z > 0) then nx, ny = 0.5, 0.5 end
depth = vp.Z > 0 and vp.Z or depth
end)
return nx, ny, depth
end
local function firePair(h)
RunService.RenderStepped:Wait()  -- one frame sync for camera
local sx, sy, sz = viewportArgs(h.root)
-- reuse same viewport args (player isn't moving during harvest)
if not fireServer("Foraging","start_harvest", h.instanceId, h.token, true, sx, sy, sz) then
return false, "start_fire_failed"
end
task.wait((h.hold or 1.75) + CONFIG.HoldMargin)
if not fireServer("Foraging","finish_harvest", h.instanceId, h.token, true, sx, sy, sz) then
return false, "finish_fire_failed"
end
task.wait(CONFIG.FinishRepeatGap)
fireServer("Foraging","finish_harvest", h.instanceId, h.token, true, sx, sy, sz)
task.wait(CONFIG.ResultWait)
return true, "remote_pair"
end
local function isInvalidMovement(res)
return res and res.ok == false and type(res.msg) == "string"
and res.msg:lower():find("invalid movement path", 1, true)
end
local function renew(dwell)
if toast then toast("anti-cheat: settling to renew", "warn") end
if log   then log("invalidation suspected -> re-walk + longer settle", "warn") end
local r = hrp(); if r then groundSnap(r.Position, r.Position.Y) end
task.wait(dwell or CONFIG.RenewDwell)
State.invalid = false
end
------------------------------------------------------------------ selection
local function isTypeLocked(id) return (minMasteryFor(id) or 0) > foragingLevel() end
local function knownSorted() local t = {}; for id in pairs(knownTypes) do t[#t+1] = id end; table.sort(t); return t end
local function nearestInstanceOf(id)
local list = State.byType[id]; if not list then return nil end
local best, bd = nil, 1e9
for _, h in ipairs(list) do
local d = distTo(h)
if d < bd then bd = d; best = h end
end
return best
end
-- returns ALL herbs from selected types, sorted by distance to player (nearest first)
local function selectedHerbsSorted()
local t = {}
for id in pairs(State.selectedTypes) do
if not isTypeLocked(id) and State.byType[id] then
for _, h in ipairs(State.byType[id]) do
local d = distTo(h)
t[#t+1] = { herb = h, d = d }
end
end
end
table.sort(t, function(a, b) return a.d < b.d end)
local out = {}
for i, v in ipairs(t) do out[i] = v.herb end
return out
end
------------------------------------------------------------------ automation
local function tryHarvest(h, walkOnly)
if log then log(("-> %s [%s]%s"):format(h.id, h.region, walkOnly and " (walk)" or ""), "info") end
if not Move.approach(h, walkOnly) then
return false
end
for tries = 1, CONFIG.MaxRetries + 1 do
if not State.running then return false end
-- check if already collected by a previous finish_harvest event
if Harvest.collected[h.instanceId] then
State.stats.harvested = State.stats.harvested + 1
if log then log("   already collected (ok)", "good") end
task.wait(CONFIG.HarvestCooldown)
return true
end
-- check if model still exists (already collected or despawned)
if not (h.model and h.model.Parent) then
-- if we got a result, it was collected — count as ok
local alreadyDone = Harvest.lastResult[h.instanceId]
if alreadyDone and alreadyDone.ok == true then
State.stats.harvested = State.stats.harvested + 1
return true
end
-- herb gone but no result — silently skip (was likely collected by finish_harvest)
return false
end
if not withinRange(h) then walkTo(h.pos()); groundSnap(h.pos(), h.pos().Y); task.wait(0.3) end
Harvest.lastResult[h.instanceId] = nil
Harvest.collected[h.instanceId] = nil
State.invalid = false
local ok, how = firePair(h)
local res = Harvest.lastResult[h.instanceId]
-- check if finish_harvest was received (collected)
if Harvest.collected[h.instanceId] or (res and res.ok == true) then
State.stats.harvested = State.stats.harvested + 1
State.invalid = false
local n = "1"
if res and res.msg then n = tostring(res.msg):match("(%d+)x") or "1" end
if toast then toast(("+%s %s"):format(n, h.id), "good") end
if log   then log(("   ok (%sx)"):format(n), "good") end
task.wait(CONFIG.HarvestCooldown)
return true
end
-- harvest result: check why
if res and res.ok == false then
local m = tostring(res.msg)
if log then log(("   server: %s"):format(m), "warn") end
-- mastery lock = can't harvest this type
if m:lower():find("mastery", 1, true) then
return false
end
-- invalid movement = anti-cheat, just re-fire (don't count as fail)
if isInvalidMovement(res) then
if log then log("   invalid movement, retrying...", "info") end
State.invalid = false  -- clear, just retry
task.wait(0.5)
-- don't increment failed, just loop again
-- re-walk to position if needed
if not withinRange(h) then
walkTo(h.pos()); groundSnap(h.pos(), h.pos().Y); task.wait(0.3)
end
else
-- other server error = retry
end
else
-- no result = timeout, retry
end
task.wait(0.25)
end
return false
end
-- automation-specific connections (killed instantly on stop)
local _autoBag = {}
local function trackAuto(c) if c then _autoBag[#_autoBag+1] = c end; return c end
local function killAuto()
State.running = false
for _, c in ipairs(_autoBag) do pcall(function() c:Disconnect() end) end
_autoBag = {}
end
local function startAuto()
if State.running then return end
if #selectedHerbsSorted() == 0 then
if toast then toast("select a herb type first", "warn") end
if log   then log("no harvestable types selected", "warn") end
return
end
State.running = true
persistAutoStart(true)
saveSettings()
if toast then toast("automation started", "good") end
if log   then log("automation START  mode=" .. State.mode, "good") end
trackAuto(task.spawn(function()
while alive and State.running do
local herbs = selectedHerbsSorted()
local progressed = false
if #herbs > 0 then
-- lock onto the nearest herb — don't switch until collected or failed
local target = herbs[1]
if target and target.model and target.model.Parent then
if State.invalid then renew() end
local result = tryHarvest(target, false)
progressed = true
-- only re-sort after current herb is done (collected or failed)
end
end
task.wait(progressed and 0.15 or CONFIG.IdleWait)
end
State.running = false
if updateStatus then updateStatus() end
if toast then toast("automation stopped", "info") end
if log   then log("automation STOP", "info") end
end))
end
local function stopAuto()
killAuto()
persistAutoStart(false)
saveSettings()
end

function openSettings()

function closeSettings()

if gui.settingsPanel then
-- smooth fade out + slide down
local tw = TweenSvc:Create(gui.settingsPanel, TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.In),

tw:Play(); tw.Completed:Connect(function()
settingsFrame.Visible = false
gui.settingsPanel.Position = UDim2.new(0,0,0,0)
gui.settingsPanel.BackgroundTransparency = 0
-- restore main content with fade in

if harvestContent then harvestContent.Visible = true end
if listFrame then listFrame.Visible = true end
if emptyState and State.lastScanHerbs == 0 then emptyState.Visible = true end
end)
else
settingsFrame.Visible = false
if harvestSidebar then harvestSidebar.Visible = true end
if harvestContent then harvestContent.Visible = true end
if listFrame then listFrame.Visible = true end
if emptyState and State.lastScanHerbs == 0 then emptyState.Visible = true end
end
end

local ROMAN = {[1]="I",[2]="II",[3]="III",[4]="IV",[5]="V",[6]="VI",[7]="VII",[8]="VIII",
[9]="IX",[10]="X",[11]="XI",[12]="XII",[13]="XIII",[14]="XIV",[15]="XV",
[16]="XVI",[17]="XVII",[18]="XVIII",[19]="XIX",[20]="XX"}
local function romanVal(t)
local map = {I=1,V=5,X=10,L=50,C=100,D=500,M=1000}
local total, prev = 0, 0
for i = #t, 1, -1 do
local v = map[t:sub(i,i)] or 0
total = total + (v < prev and -v or v); prev = v
end
return total
end
local function parseStage(stage)
if type(stage) ~= "string" then return "", nil end
local prefix, tail = stage:match("^(.-)%s+([IVXLCDM]+)$")
if not prefix then return stage, nil end
return prefix, romanVal(tail)
end
local function normPath(p)
if type(p) ~= "string" then return nil end
local t = p:match("^%s*(.-)%s*$") or p
if t == "" or t:lower() == "none" then return nil end
return t
end
local function fmtN(n)
if n == nil or n ~= n then return "—" end
local a = math.abs(n)
if a >= 1e9 then return ("%.2fB"):format(n/1e9) end
if a >= 1e6 then return ("%.2fM"):format(n/1e6) end
if a >= 1e3 then return ("%.2fK"):format(n/1e3) end
return tostring(math.floor(n + 0.5))
end
local function fmtRate(n)
if n == nil or n ~= n then return "—" end
return fmtN(n) .. "/s"
end
local function fmtT(s)
if s == nil or s ~= s or s >= 1e9 then return "—" end
if s <= 0 then return "0s" end
local h = math.floor(s / 3600)
local m = math.floor((s % 3600) / 60)
local sc = math.floor(s % 60)
if h > 0 then return ("%dh %dm %ds"):format(h, m, sc) end
if m > 0 then return ("%dm %ds"):format(m, sc) end
return ("%ds"):format(sc)
end
local function pct(x) return ("%.1f%%"):format((x or 0) * 100) end

-- chance model (quadratic concave fit)
local function solve3(m, t)
local M = {
{m[1][1],m[1][2],m[1][3],t[1]},
{m[2][1],m[2][2],m[2][3],t[2]},
{m[3][1],m[3][2],m[3][3],t[3]},
}
for ccol = 1, 3 do
local piv, best = ccol, math.abs(M[ccol][ccol])
for r = ccol+1, 3 do
local a = math.abs(M[r][ccol])
if a > best then best = a; piv = r end
end
if best < 1e-12 then return nil end
if piv ~= ccol then M[ccol], M[piv] = M[piv], M[ccol] end
local d = M[ccol][ccol]
for r = 1, 3 do
if r ~= ccol then
local f = M[r][ccol] / d
for c2 = ccol, 4 do M[r][c2] = M[r][c2] - f * M[ccol][c2] end
end
end
end
return { M[1][4]/M[1][1], M[2][4]/M[2][2], M[3][4]/M[3][3] }
end
local function fitChance()
local n = #cultSamples
if n < 2 then return nil end
local S0,S1,S2,S3,S4 = 0,0,0,0,0
local T0,T1,T2 = 0,0,0
for i = 1, n do
local r, ch = cultSamples[i].r, cultSamples[i].ch
local r2 = r*r; local r3 = r2*r; local r4 = r3*r
S0 = S0+1; S1 = S1+r; S2 = S2+r2; S3 = S3+r3; S4 = S4+r4
T0 = T0+ch; T1 = T1+r*ch; T2 = T2+r2*ch
end
local lin = nil
local det2 = S0*S2 - S1*S1
if math.abs(det2) > 1e-12 then
lin = { (T0*S2 - T1*S1)/det2, (S0*T1 - S1*T0)/det2 }
end
local quad = nil
if n >= 6 then
quad = solve3({{S0,S1,S2},{S1,S2,S3},{S2,S3,S4}}, {T0,T1,T2})
end
return { n = n, lin = lin, quad = quad }
end
local function useQuad(fit)
if not fit then return false end
return (fit.quad ~= nil and fit.n >= 10 and math.abs(fit.quad[3]) > 1e-7)
end
local function chanceAtCost(fit)
if not fit then return nil, "none" end
if useQuad(fit) then
local a,b,c = fit.quad[1], fit.quad[2], fit.quad[3]
return math.min(1, math.max(0, a+b+c)), "quad"
elseif fit.lin then
local a,b = fit.lin[1], fit.lin[2]
return math.min(1, math.max(0, a+b)), "linear"
end
return nil, "none"
end

-- subscribe to Cultivation.state_updated
local cultConn = nil
local function installCultState()
local folder = Replicated:FindFirstChild("Network")
if not folder then return false end
local cultFolder = folder:FindFirstChild("Cultivation")
if not cultFolder then return false end
local remote = cultFolder:FindFirstChild("state_updated")
if not (remote and remote:IsA("RemoteEvent")) then return false end
cultConn = remote.OnClientEvent:Connect(function(payload)
if type(payload) ~= "table" then return end
local realNow = normPath(payload.ActivePath) or normPath(payload.GainPath)
if realNow and payload[realNow] then cultActive = realNow end
local active = (realNow and payload[realNow]) and realNow
or (payload[cultActive] and cultActive)
or (payload.Qi and "Qi") or (payload.Body and "Body") or cultActive
cultActive = active
local node = payload[active]
if not node then cultState = payload; return end
local cur = node.Current or 0
local cost = node.BreakthroughCost or 0
local ch = node.BreakthroughChance or 0
local now = tick()
if active ~= cultLastPath or (cost and cultLastCost and math.abs(cost - cultLastCost) > 1) then
cultSamples = {}; cultLastPath = active; cultLastCost = cost; cultLastPushedCur = nil
cultObs.rate, cultObs.cslope = nil, nil
cultObsLastT, cultObsLastCur, cultObsLastCh = nil, nil, nil
end
cultLastCost = cost
if cur and cost and cost > 0 and ch then
if cultLastPushedCur == nil or math.abs(cur - cultLastPushedCur) > 0.5 then
local r = cur / cost
if r == r and r >= 0 then  -- skip NaN
table.insert(cultSamples, { r = r, ch = ch })
cultLastPushedCur = cur
while #cultSamples > cultSampleCap do table.remove(cultSamples, 1) end
end
end
end
if cultObsLastT and cur and cultObsLastCur then
local dt = now - cultObsLastT
if dt > 0 then
local ir = (cur - cultObsLastCur) / dt
cultObs.rate = cultObs.rate and (cultObs.rate*0.7 + ir*0.3) or ir
if ch and cultObsLastCh then
local ic = (ch - cultObsLastCh) / dt
cultObs.cslope = cultObs.cslope and (cultObs.cslope*0.7 + ic*0.3) or ic
end
end
end
cultObsLastT, cultObsLastCur, cultObsLastCh = now, cur, ch
cultState = payload
end)
return true
end

local function armRejoin()

repeat task.wait(1) until game:IsLoaded()

if request then local o,r=pcall(function() return request({Url=u,Method="GET"}) end) if o and r and r.Body then return r.Body end end
if http_request then local o,r=pcall(function() return http_request({Url=u,Method="GET"}) end) if o and r and r.Body then return r.Body end end
if http and http.request then local o,r=pcall(function() return http.request({Url=u,Method="GET"}) end) if o and r and r.Body then return r.Body end end
if syn and syn.request then local o,r=pcall(function() return syn.request({Url=u,Method="GET"}) end) if o and r and r.Body then return r.Body end end
local o,b=pcall(function() return game:HttpGet(u) end) if o and b then return b end

if s then local f=loadstring(s); if f then f() end end
]==], placeId, FK_URL)

if not q then pcall(function() if syn then q = syn.queue_on_teleport end end) end
if not q then pcall(function() q = queueonteleport end) end
if q then pcall(q, boot) end
end

function destroy()
if not alive then return end
alive = false
State.running = false
stopCapture(true)
setScanning(false)
for _, c in ipairs(bag) do pcall(function() c:Disconnect() end) end
for m, rec in pairs(tracked) do detachRecord(rec) end
tracked = {}
pcall(function()
local c = cam()
if c and c.CameraType == Enum.CameraType.Scriptable then c.CameraType = Enum.CameraType.Custom end
end)
if guiRoot    then pcall(function() guiRoot:Destroy() end) end
if toastHolder then pcall(function() toastHolder:Destroy() end) end
if getgenv then getgenv().ForageKit = nil end
end
------------------------------------------------------------------ input
track(UserInput.InputBegan:Connect(function(inp, gpe)
if gpe or not alive then return end
if inp.UserInputType ~= Enum.UserInputType.Keyboard then return end
if capturing then
if inp.KeyCode == Enum.KeyCode.Escape then
stopCapture(true)
else
local key = inp.KeyCode
local act = capturing.action
for other, btn in pairs(bindButtons) do
if other ~= act and Binds[other] == key then
Binds[other] = nil
btn.Text = "—"
end
end
Binds[act] = key
stopCapture(false)
saveSettings()
end
return
end
if inp.KeyCode == Binds.toggle then
toggle()
elseif inp.KeyCode == Binds.start then
startAuto()
elseif inp.KeyCode == Binds.stop then
stopAuto()
if toast then toast("stopped", "info") end
end
end))
