local function F(t,...) local a={...} for i=1,select('#',...) do if a[i]==nil then a[i]="nil" end end return t:format(unpack(a)) end
local function log(s) lc=lc+1; out[lc]=s end

-- walk pacing: yields every 200 instances, progress print every 5000
local function walkPace(ctr,t0)
    ctr[1]=ctr[1]+1
    local n=ctr[1]
    if n%CAP.yieldEvery==0 then yield() end
    if n%CAP.progressEvery==0 then print(F("[G] walked %d (%ds)",n,math.floor(tick()-t0))) end
end

-- writefile helper: tries GatherV4/<name>, falls back to flat <name>
local function wfileRaw(name,content)
    if not writefile then return nil end
    local ok,err = pcall(function() writefile("GatherV4/"..name,content) end)
    if ok then return "GatherV4/"..name end
    local ok2,err2 = pcall(function() writefile(name,content) end)
    if ok2 then return name end
    log(F("[FAIL] write %s: %s",name,tostring(err2)))
    return nil
end

-- chunked write: keeps big outputs under executor writefile caps (~2MB)
local function chunkedWrite(path, content, maxBytes)
    if #content<=maxBytes then return wfileRaw(path,content) end
    local head = "--[TRUNCATED to "..maxBytes.." bytes for writefile cap]--\n"
    return wfileRaw(path,head..content:sub(1,maxBytes-#head-40))
end

-- safe string -> lua literal (for generated clone.lua); escapes control bytes too
local function luaStr(s)
    local parts={}
    for i=1,#s do
        local b=s:byte(i)
        if b==92 then parts[#parts+1]="\\\\"
        elseif b==34 then parts[#parts+1]='\\"'
        elseif b==10 then parts[#parts+1]="\\n"
        elseif b==13 then parts[#parts+1]="\\r"
        elseif b==9 then parts[#parts+1]="\\t"
        elseif b<32 then parts[#parts+1]=string.format("\\%d",b)
        else parts[#parts+1]=string.char(b) end
    end
    return '"'..table.concat(parts)..'"'
end

-- report value serializer with type tags
local function serT(v)
    local t=type(v)
    if t=="nil" then return "nil"
    elseif t=="string" then return '"'..v:sub(1,120)..'"'
    elseif t=="number" or t=="boolean" then return tostring(v)
    elseif t=="table" then return F("{tbl:%d}",tlen(v))
    elseif t=="Instance" then return "[Inst] "..fp(v)
    else
        local s=tostring(v)
        if #s>120 then s=s:sub(1,120).."..." end
        return F("<%s>%s",t,s)
    end
end

-- shallow table serialization for spy (cap 12 keys)
local function serTbl(t)
    local parts={}
    local n=0
    for k,v in pairs(t) do
        n=n+1
        if n>12 then parts[#parts+1]="..."; break end
        parts[#parts+1]=F("%s=%s",tostring(k),serT(v))
    end
    return "{"..table.concat(parts,", ").."}"
end

local svcCache = {}
local function getService(n)
    local got = svcCache[n]
    if got ~= nil then return got or nil end
    local ok,s = pcall(function() return game:GetService(n) end)
    if not ok or not s then s = nil end
    svcCache[n] = s or false
    return s
end

local function nilInstances()
    local list = {}
    if getnilinstances then
        local ok,ni = pcall(getnilinstances)
        if ok and type(ni)=="table" then
            for _,v in ipairs(ni) do list[#list+1]=v end
        end
    end
    return list
end

-- roots for a service: the real service + matching nil instances
local function serviceRoots(name)
    local roots = {}
    local seen = {}
    local svc = getService(name)
    if svc then roots[#roots+1]=svc; seen[svc]=true end
    if getnilinstances then
        local ok,ni = pcall(getnilinstances)
        if ok and type(ni)=="table" then
            for _,v in ipairs(ni) do
                if not seen[v] then
                    local ok2,p = pcall(function() return v:GetFullName() end)
                    local match = false
                    if ok2 and p then
                        if p==name or p:sub(1,#name+1)==name.."." then
                            match = true
                        elseif (name=="ServerScriptService" or name=="ServerStorage") and not p:find(".",1,true) then
                            match = true -- unparented server instance
                        end
                    end
                    if match then roots[#roots+1]=v; seen[v]=true end
                end
            end
        end
    end
    return roots
end

-- flattened descendants of a service incl. nil-instance fallback
local function getAllUnder(name)
    local list = {}
    local svc = getService(name)
    if svc then
        local ok,ds = pcall(function() return svc:GetDescendants() end)
        if ok and ds then for _,v in ipairs(ds) do list[#list+1]=v end end
    end
    local ni = nilInstances()
    for _,v in ipairs(ni) do
        local ok3,p = pcall(function() return v:GetFullName() end)
        if ok3 and p then
            if p==name or p:sub(1,#name+1)==name.."." then
                list[#list+1]=v
            elseif (name=="ServerScriptService" or name=="ServerStorage") and not p:find(".",1,true) then
                list[#list+1]=v
            end
        end
    end
    return list
end

-- resolve a full path like "Workspace.Foraging.Herb_1" against the live tree
local function resolvePath(p)
    if not p or p=="" then return nil end
    local segs={}
    for s in p:gmatch("[^.]+") do segs[#segs+1]=s end
    local cur=getService(segs[1])
    if not cur then return nil end
    for i=2,#segs do
        cur=cur:FindFirstChild(segs[i])
        if not cur then return nil end
    end
    return cur
end

local scripts = {}

local scrIdx = {}
local function sanitizeName(n)
    n=n:gsub("[\\/:*?\"<>|]","_")
    n=n:gsub("^[%.%s]+","")
    if #n>40 then n=n:sub(1,40) end
    if n=="" then n="script" end
    return n
end

local savedSrc = {} -- fp -> file name, avoid duplicate saves
local decompN, bcN, failN, skipN = 0,0,0,0

local function saveScriptSource(scr,label)
    local p = fp(scr)
    if savedSrc[p] then return savedSrc[p] end
    local fname = sanitizeName(sg(scr,"Name") or "script").."-"..(#scrIdx+1)..".lua"
    local ok,src = pcall(function()
        if decompile then return decompile(scr) end
        return nil
    end)
    if ok and type(src)=="string" and #src>10 then
        if #src>CAP.scriptBytes then
            skipN=skipN+1
            log(F("  [%s] %s [SKIP >2MB]",label,p))
            return nil
        end
        local wp = wfileRaw("scripts/"..fname,src)
        savedSrc[p]=wp or fname
        scrIdx[#scrIdx+1]=F("%3d | %-60s | %s | decompiled | %d bytes",#scrIdx+1,wp or fname,p,#src)
        log(F("  [%s] %s -> %s (%d bytes)",label,p,wp or fname,#src))
        decompN=decompN+1
        return wp or fname
    end
    local ok2,bc = pcall(function()
        if getscriptbytecode then return getscriptbytecode(scr) end
        return nil
    end)
    if ok2 and type(bc)=="string" and #bc>0 then
        local strs={}
        for s in bc:gmatch("[\32-\126]{6,}") do strs[#strs+1]=s end
        local content = F("--[[ BYTECODE STRINGS %s ]]--\n%s",p,table.concat(strs,"\n"))
        if #content>CAP.scriptBytes then
            skipN=skipN+1
            log(F("  [%s] %s [SKIP >2MB]",label,p))
            return nil
        end
        local wp = wfileRaw("scripts/"..fname,content)
        savedSrc[p]=wp or fname
        scrIdx[#scrIdx+1]=F("%3d | %-60s | %s | bytecode | %d bytes",#scrIdx+1,wp or fname,p,#content)
        log(F("  [%s] %s -> %s [bytecode strings, %d found]",label,p,wp or fname,#strs))
        bcN=bcN+1
        return wp or fname
    end
    failN=failN+1
    log(F("  [%s] %s [FAIL]",label,p))
    return nil
end

for i,scr in ipairs(scripts) do
    if i>CAP.scripts then log(F("[CAP] scripts capped at %d",CAP.scripts)); break end
    if i%25==0 then print(F("[G] scripts %d/%d",i,#scripts)) end
    if i%10==0 then yield() end
    saveScriptSource(scr,"script")
end
log(F("Summary: decompiled=%d bytecode=%d failed=%d skipped=%d",decompN,bcN,failN,skipN))

            if vl==t then hit=true; break end

            modSeen[v]=true
            modCount=modCount+1
            if modCount>CAP.modules then log(F("[CAP] module requires capped at %d",CAP.modules)); break end
            local p = fp(v)
            log(F("  [%d/%d] %s",modCount,CAP.modules,p))
            local ok,mod = pcall(require,v)
            if not ok then
                log("    require failed: "..tostring(mod))
            else
                log(F("    type: %s",type(mod)))
                if type(mod)=="table" then
                    local m=0
                    for kk,vv in pairs(mod) do
                        m=m+1
                        if m<=CAP.modKeys then log(F("      %s = %s",tostring(kk),type(vv))) end
                    end
                    if m>CAP.modKeys then log(F("      ...+%d more",m-CAP.modKeys)) end
                    -- table of functions -> try decompile source save
                    local fnCount=0
                    for _,vv in pairs(mod) do
                        if type(vv)=="function" then fnCount=fnCount+1 end
                    end
                    if fnCount>0 then
                        local ok2,src = pcall(function()
                            if decompile then return decompile(v) end
                            return nil
                        end)
                        if ok2 and type(src)=="string" and #src>10 and #src<=CAP.scriptBytes then
                            local wp = saveScriptSource(v,"module")
                            if wp then log(F("    source -> %s",wp)) end
                        end
                    end
                end
            end
        end
    end
end

sec("Values / Positions / Attributes")

local valOut = {}
local posOut = {}
local attrOut = {}

local VALUE_CLASSES = {
    IntValue=true, NumberValue=true, StringValue=true, BoolValue=true,
    ObjectValue=true, Vector3Value=true, CFrameValue=true,
    BrickColorValue=true, Color3Value=true,
}

local vCount, pCount, aCount = 0,0,0
local vTotal, pTotal, aTotal = 0,0,0
local vCapHit, pCapHit, aCapHit = false,false,false
local htypes = {}
local herbs = {}

local function interestingPart(d)
    -- cheap checks first: name keywords, then attributes, then ancestor folders
    local nl = (sg(d,"Name") or ""):lower()
    for _,kw in ipairs(PART_KW) do
        if nl:find(kw,1,true) then return true end
    end
    local ok,at = pcall(function() return d:GetAttributes() end)
    if ok and at and tlen(at)>0 then return true end
    local cur = d
    for i=1,6 do
        local ok2,par = pcall(function() return cur.Parent end)
        if not ok2 or not par then break end
        cur = par
        local c2 = cn(cur)
        if c2=="Folder" or c2=="Model" then
            local an = (sg(cur,"Name") or ""):lower()
            for _,kw in ipairs(PART_FOLDER_KW) do
                if an:find(kw,1,true) then return true end
            end
        end
        if c2=="Workspace" then break end
    end
    return false
end

local function processInstance(d)
    local c = cn(d)
    if VALUE_CLASSES[c] then
        vTotal=vTotal+1
        if not vCapHit then
            local v = sg(d,"Value")
            valOut[#valOut+1]=F("[%s] %s = %s",c,fp(d),serT(v))
            vCount=vCount+1
            if vCount>=CAP.vals then
                vCapHit=true
                valOut[#valOut+1]=F("...  [CAP] values capped at %d",CAP.vals)
            end
        end
    end
    if c=="BasePart" then
        pTotal=pTotal+1
        if not pCapHit and interestingPart(d) then
            local pos,size = sg(d,"Position"), sg(d,"Size")
            local ps = pos and F("(%s, %s, %s)",serNum(pos.X),serNum(pos.Y),serNum(pos.Z)) or "?"
            local ss = size and F("(%s, %s, %s)",serNum(size.X),serNum(size.Y),serNum(size.Z)) or "?"
            posOut[#posOut+1]=F("[%s] %s  Pos=%s  Size=%s",c,fp(d),ps,ss)
            pCount=pCount+1
            if pCount>=CAP.positions then
                pCapHit=true
                posOut[#posOut+1]=F("...  [CAP] positions capped at %d",CAP.positions)
            end
        end
    end
    local ok,at = pcall(function() return d:GetAttributes() end)
    if ok and at and tlen(at)>0 then
        if at.HerbId then
            herbs[#herbs+1]=d
            local hid = at.HerbId
            htypes[hid]=(htypes[hid] or 0)+1
        end
        local interesting=false
        for k,_ in pairs(at) do
            local kl=k:lower()
            for _,kw in ipairs(KEYWORDS) do
                if kl:find(kw,1,true) then interesting=true; break end
            end
            if interesting then break end
        end
        if interesting then

-- property snapshot: getproperties (adapted at runtime) or curated fallback
local function collectPropNames(inst)
    if not getproperties then return nil end
    local ok,props = pcall(getproperties,inst)
    if not ok or type(props)~="table" then return nil end
    local n = tlen(props)
    if n==0 then return nil end
    local names={}
    if props[1]~=nil then
        for i=1,n do
            local e=props[i]
            if type(e)=="table" then
                local nm = e.Name or e.name
                if type(nm)=="string" then names[#names+1]=nm end
                if #names>=100 then break end
            end
        end
    else
        for k in pairs(props) do
            if type(k)=="string" then names[#names+1]=k end
            if #names>=100 then break end
        end
    end
    if #names==0 then return nil end
    return names
end

local function curatedProps(inst)
    local list={}
    if isA(inst,"BasePart") then
        for _,p in ipairs({"Position","Orientation","Size","Anchored","CanCollide","Transparency","Material","Color","CastShadow"}) do list[#list+1]=p end
    elseif isA(inst,"GuiObject") then
        for _,p in ipairs({"Position","Size","AnchorPoint","BackgroundColor3","BackgroundTransparency","TextColor3","Text","Visible","ZIndex"}) do list[#list+1]=p end
        if isA(inst,"TextLabel") or isA(inst,"TextButton") then
            for _,p in ipairs({"TextScaled","FontSize","TextWrapped"}) do list[#list+1]=p end
        end
    end
    local c=cn(inst)
    if VALUE_CLASSES[c] then
        list[#list+1]="Value"
    elseif c=="Model" then
        list[#list+1]="PrimaryPart"
    end
    return list
end

-- prop value -> clone-lua serialization ("TYPE|data")
local function serPropVal(name,v)
    local t=type(v)
    if t=="number" then return "N|"..serNum(v)
    elseif t=="string" then
        if #v>200 then v=v:sub(1,200) end
        return "S|"..v
    elseif t=="boolean" then return "B|"..(v and "1" or "0")
    elseif t=="Vector3" then return F("V3|%s,%s,%s",serNum(v.X),serNum(v.Y),serNum(v.Z))
    elseif t=="Vector2" then return F("V2|%s,%s",serNum(v.X),serNum(v.Y))
    elseif t=="Color3" then return F("C3|%s,%s,%s",serNum(v.R),serNum(v.G),serNum(v.B))
    elseif t=="UDim2" then return F("UD2|%s,%s,%s,%s",serNum(v.X.Scale),serNum(v.X.Offset),serNum(v.Y.Scale),serNum(v.Y.Offset))
    elseif t=="UDim" then return F("UD|%s,%s",serNum(v.Scale),serNum(v.Offset))
    elseif t=="CFrame" then
        local ok,x,y,z,a,b,c,d,e,f,g,h,i = pcall(function() return v:GetComponents() end)
        if ok and x then
            return F("CF|%s,%s,%s,%s,%s,%s,%s,%s,%s,%s,%s,%s",serNum(x),serNum(y),serNum(z),serNum(a),serNum(b),serNum(c),serNum(d),serNum(e),serNum(f),serNum(g),serNum(h),serNum(i))
        end
        return nil
    elseif t=="EnumItem" then
        local s=tostring(v)
        if s:sub(1,5)=="Enum." then return "E|"..s end
        return nil
    elseif t=="Instance" then
        if name=="PrimaryPart" then
            local p=fp(v)
            if p and p~="" then return "I|"..p end
        end
        return nil
    end
    return nil
end

-- full = record's own path, par = parent's path (carried on the DFS stack,
-- never derived by string-splitting names, so names containing "." stay safe)
local function addRecord(inst,full,par)
    if recCount>=CAP.clone then return end
    local nm = sg(inst,"Name")
    if not nm or nm=="" then return end
    local props={}
    local names = collectPropNames(inst) or curatedProps(inst)
    for _,pn in ipairs(names) do
        if not (pn=="Parent" or pn=="Name" or pn=="ClassName") then
            local ok,pv = pcall(function() return inst[pn] end)
            if ok then
                local s=serPropVal(pn,pv)
                if s then
                    props[#props+1]={pn,s}
                    if #props>=100 then break end
                end
            end
        end
    end
    records[#records+1]={cn=cn(inst),n=nm,p=par,props=props}
    recCount=recCount+1
end

local depthSkipped = 0
local function walkCloneRoot(root,recordRoot,rootName)
    local stack={}
    if recordRoot then
        stack[#stack+1]={root,1,""} -- nil-instance root: recorded itself, parent ""
    else
        local ok,kids=pcall(function() return root:GetChildren() end)
        if ok and kids then
            for i=#kids,1,-1 do stack[#stack+1]={kids[i],1,rootName} end
        end
    end
    while #stack>0 and recCount<CAP.clone do
        local e=table.remove(stack)
        local node,dep,pp = e[1],e[2],e[3]
        if dep<=CAP.cloneDepth then
            local c=cn(node)
            local nm = sg(node,"Name")
            local full = nm and (pp=="" and nm or pp.."."..nm) or nil
            if not skipSet[c] and full then
                addRecord(node,full,pp)
            end
            local ok,kids=pcall(function() return node:GetChildren() end)
            if ok and kids then
                for i=#kids,1,-1 do stack[#stack+1]={kids[i],dep+1,full or pp} end
            end
        else
            depthSkipped=depthSkipped+1
        end
        walkPace(ctr5,t05)
    end
end
