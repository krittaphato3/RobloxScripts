local function loadSettings()
    local ok, exists = pcall(isfile, SETTINGS_FILE)
    if not (ok and exists) then return end
    local ok2, txt = pcall(readfile, SETTINGS_FILE)
    if not (ok2 and type(txt) == "string" and #txt > 0) then return end
    local ok3, fn = pcall(loadstring, txt)
    if not (ok3 and type(fn) == "function") then return end
    local ok4, t = pcall(fn)
    if not (ok4 and type(t) == "table") then return end
    if type(t.key) == "string" then SET.key = t.key end
    if t.keyEnabled == true or t.keyEnabled == false then SET.keyEnabled = t.keyEnabled end
    if type(t.w) == "number" and t.w >= 300 and t.w <= 760 then SET.w = math.floor(t.w) end
    if type(t.h) == "number" and t.h >= 280 and t.h <= 860 then SET.h = math.floor(t.h) end
    if type(t.x) == "number" then SET.x = math.floor(t.x) end
    if type(t.y) == "number" then SET.y = math.floor(t.y) end
    if t.mode == "auto" or t.mode == "quad" or t.mode == "linear" then SET.mode = t.mode end
    if type(t.win) == "number" and t.win >= 20 and t.win <= 180 then SET.win = math.floor(t.win) end
    -- v2.4
    if t.autoBt == true or t.autoBt == false then SET.autoBt = t.autoBt end
    if t.autoBtBody == true or t.autoBtBody == false then SET.autoBtBody = t.autoBtBody end
    if type(t.minChance) == "number" then SET.minChance = math.clamp(math.floor(t.minChance), 0, 99) end

local function addCommas(n)
    local s = string.format("%d", math.floor(n + 0.5))
    local rev = s:reverse():gsub("(%d%d%d)", "%1,"):gsub(",$", "")
    return rev:reverse()
end
local function fmtN(n)
    if n == nil or n ~= n then return "—" end
    local a = math.abs(n)
    if a >= 1e9 then return string.format("%.2fB (%s)", n/1e9, addCommas(n)) end
    if a >= 1e6 then return string.format("%.2fM (%s)", n/1e6, addCommas(n)) end
    if a >= 1e3 then return string.format("%.2fK (%s)", n/1e3, addCommas(n)) end
    return addCommas(n)
end
local function fmtRate(n)
    if n == nil or n ~= n then return "—" end
    return fmtN(n) .. "/s"
end
local function fmtT(s)
    if s == nil or s ~= s or s >= 1e9 then return "—" end
    if s <= 0 then return "0s" end
    local d  = math.floor(s / 86400)
    local h  = math.floor((s % 86400) / 3600)
    local m  = math.floor((s % 3600) / 60)
    local sc = math.floor(s % 60)
    if d > 0 then return string.format("%dd %dh %dm", d, h, m) end
    if h > 0 then return string.format("%dh %dm %ds", h, m, sc) end
    if m > 0 then return string.format("%dm %ds", m, sc) end
    return string.format("%ds", sc)
end

local function bar(frac, width)
    width = width or 22
    if frac ~= frac or frac == nil then frac = 0 end
    if frac < 0 then frac = 0 end
    if frac > 1 then frac = 1 end
    local f = math.floor(frac * width + 0.5)
    return string.rep("█", f) .. string.rep("░", width - f)
end

local ROMAN = {[1]="I",[2]="II",[3]="III",[4]="IV",[5]="V",[6]="VI",[7]="VII",[8]="VIII",
    [9]="IX",[10]="X",[11]="XI",[12]="XII",[13]="XIII",[14]="XIV",[15]="XV",
    [16]="XVI",[17]="XVII",[18]="XVIII",[19]="XIX",[20]="XX"}
local function romanVal(t)
    local map = {I=1,V=5,X=10,L=50,C=100,D=500,M=1000}
    local total, prev = 0, 0
    for i = #t, 1, -1 do
        local v = map[t:sub(i,i)] or 0
        total = total + (v < prev and -v or v); prev = v
    end
    return total
end
local function parseStage(stage)
    if type(stage) ~= "string" then return "", nil end
    local prefix, tail = stage:match("^(.-)%s+([IVXLCDM]+)$")
    if not prefix then return stage, nil end
    return prefix, romanVal(tail)
end

-- ═══════════════════════════════════════════════════════════
-- CHANCE MODEL  (quadratic concave fit over live samples)
-- ═══════════════════════════════════════════════════════════
local function solve3(m, t)
    local M = {
        {m[1][1],m[1][2],m[1][3],t[1]},
        {m[2][1],m[2][2],m[2][3],t[2]},
        {m[3][1],m[3][2],m[3][3],t[3]},
    }
    for ccol = 1, 3 do
        local piv, best = ccol, math.abs(M[ccol][ccol])
        for r = ccol+1, 3 do
            local a = math.abs(M[r][ccol])
            if a > best then best = a; piv = r end
        end
        if best < 1e-12 then return nil end
        if piv ~= ccol then M[ccol], M[piv] = M[piv], M[ccol] end
        local d = M[ccol][ccol]
        for r = 1, 3 do
            if r ~= ccol then
                local f = M[r][ccol] / d
                for c2 = ccol, 4 do M[r][c2] = M[r][c2] - f * M[ccol][c2] end
            end
        end
    end
    return { M[1][4]/M[1][1], M[2][4]/M[2][2], M[3][4]/M[3][3] }
end
local function fitChance()
    local n = #samples
    if n < 2 then return nil end
    local S0,S1,S2,S3,S4 = 0,0,0,0,0
    local T0,T1,T2 = 0,0,0
    for i = 1, n do
        local r, ch = samples[i].r, samples[i].ch
        local r2 = r*r; local r3 = r2*r; local r4 = r3*r
        S0 = S0+1; S1 = S1+r; S2 = S2+r2; S3 = S3+r3; S4 = S4+r4
        T0 = T0+ch; T1 = T1+r*ch; T2 = T2+r2*ch
    end
    local lin = nil
    local det2 = S0*S2 - S1*S1
    if math.abs(det2) > 1e-12 then
        lin = { (T0*S2 - T1*S1)/det2, (S0*T1 - S1*T0)/det2 }
    end
    local quad = nil
    if n >= 6 then
        quad = solve3({{S0,S1,S2},{S1,S2,S3},{S2,S3,S4}}, {T0,T1,T2})
    end
    return { n = n, lin = lin, quad = quad }
end
local function useQuad(fit)
    if not fit then return false end
    if SET.mode == "quad" then return fit.quad ~= nil end
    if SET.mode == "linear" then return false end
    return (fit.quad ~= nil and fit.n >= 10 and math.abs(fit.quad[3]) > 1e-7)
end
local function chanceAtCost(fit)
    if not fit then return nil, "none" end
    if useQuad(fit) then
        local a,b,c = fit.quad[1], fit.quad[2], fit.quad[3]
        return math.min(1, math.max(0, a+b+c)), "quad"
    elseif fit.lin then
        local a,b = fit.lin[1], fit.lin[2]
        return math.min(1, math.max(0, a+b)), "linear"
    end
    return nil, "none"
end
local function vertexInfo(fit)
    if not (fit and fit.quad) then return nil end
    local a,b,c = fit.quad[1], fit.quad[2], fit.quad[3]
    if math.abs(c) < 1e-12 then return nil end
    local rv = -b/(2*c)
    return { r = rv, ch = math.min(1, math.max(0, a+b*rv+c*rv*rv)), concave = c < 0 }
end
local function rForTarget(fit, T, curR)
    if not fit then return nil, nil end
    local v = vertexInfo(fit)
    if useQuad(fit) then
        local a,b,c = fit.quad[1], fit.quad[2], fit.quad[3]
        if math.abs(c) < 1e-12 then
            if math.abs(b) < 1e-12 then return nil, v end
            return (T-a)/b, v
        end
        local disc = b*b - 4*c*(a - T)
        if disc < 0 then return nil, v end
        local sq = math.sqrt(disc)
        local r1 = (-b - sq)/(2*c)
        local r2 = (-b + sq)/(2*c)
        local cand = {}
        if r1 >= curR - 1e-9 then table.insert(cand, r1) end
        if r2 >= curR - 1e-9 then table.insert(cand, r2) end
        if #cand == 0 then return nil, v end
        table.sort(cand)
        return cand[1], v
    elseif fit.lin then
        local a,b = fit.lin[1], fit.lin[2]
        if math.abs(b) < 1e-12 then return nil, v end
        return (T-a)/b, v
    end
    return nil, v
end
local function attemptTimeFor(T, fit, curR, cur, cost, rate, chance)
    if not rate or rate <= 0 then return nil, "rate0" end
    if chance and chance >= T then
        if cost and cur >= cost then return 0, "attempt now" end
        local t = (cost and cost > 0) and ((cost - cur) / rate) or nil
        return t, "chance met now; gate = essence"
    end
    local rT, v = rForTarget(fit, T, curR)
    if not rT then
        if v and v.concave and v.r and v.r >= curR - 1e-9 and v.ch and v.ch < T then
            return nil, string.format("peaks %.0f%%", v.ch*100)
        end
        return nil, "unreachable"
    end
    local rUse = math.max(rT, 1)
    local need = rUse*cost - cur
    if need < 0 then need = 0 end
    local t = need / rate
    local tag = (rT <= 1.0001) and "ok@cost" or "overfill"
    return t, tag
end

local stateConn = nil
local function installState()
    local r = getRemote("Network.Cultivation.state_updated")
    if not (r and r:IsA("RemoteEvent")) then return false end
    stateConn = r.OnClientEvent:Connect(function(payload)   -- named param, NO varargs
        if type(payload) ~= "table" then return end
        local realNow = normPath(payload.ActivePath) or normPath(payload.GainPath)
        if realNow and payload[realNow] then lastRealPath = realNow end
        local active = (realNow and payload[realNow]) and realNow
                    or (payload[lastRealPath] and lastRealPath)
                    or (payload.Qi and "Qi") or (payload.Body and "Body") or lastRealPath
        local node = payload[active]
        if not node then return end
        local cur  = node.Current
        local cost = node.BreakthroughCost
        local ch   = node.BreakthroughChance
        local now  = tick()
        local idle = (realNow == nil)
        if active ~= lastPath or (cost and lastCost and math.abs(cost - lastCost) > 1) then
            samples = {}; lastPath = active; lastCost = cost; lastPushedCur = nil
            obs.rate, obs.cslope = nil, nil
            obsLastT, obsLastCur, obsLastCh = nil, nil, nil
        end
        lastCost = cost
        if cur and cost and cost > 0 and ch then
            if lastPushedCur == nil or math.abs(cur - lastPushedCur) > 0.5 then
                table.insert(samples, { r = cur / cost, ch = ch })
                lastPushedCur = cur
                while #samples > sampleCap do table.remove(samples, 1) end
            end
        end
        if obsLastT and cur and obsLastCur then
            local dt = now - obsLastT
            if dt > 0 then
                local ir = (cur - obsLastCur) / dt
                obs.rate = obs.rate and (obs.rate*0.7 + ir*0.3) or ir
                if ch and obsLastCh then
                    local ic = (ch - obsLastCh) / dt
                    obs.cslope = obs.cslope and (obs.cslope*0.7 + ic*0.3) or ic
                end
            end
        end
        obsLastT, obsLastCur, obsLastCh = now, cur, ch
        S, S_active, S_idle = payload, active, idle
    end)
    return true
end

        statusDot.BackgroundColor3 = gaining and C.good or (idle and C.warn or C.inkDim)
    end

    local cur   = node.Current or 0
    local cost  = node.BreakthroughCost or 0
    local gap   = math.max(0, cost - cur)
    local rate  = S[active .. "EffectiveRate"] or S[active .. "GrowthRate"] or 0
    local tCost = (rate and rate > 0 and gap > 0) and (gap / rate) or (gap <= 0 and 0 or nil)
    local chance= node.BreakthroughChance or 0
    local curR  = (cost and cost > 0) and (cur / cost) or 0
    local fit   = fitChance()
    local chCost, modelName = chanceAtCost(fit)
    local vtx   = vertexInfo(fit)

    local L = {}
    local stateWord
    if gaining then
        stateWord = col(GOOD, "MEDITATING")
    elseif idle then
        stateWord = col(WARN, "IDLE") .. col(DIM, " · not cultivating")
    else
        stateWord = col(WARN, "IDLE")
    end
    table.insert(L, heroSm(accent, active:upper() .. " PATH") .. "   " .. stateWord
        .. "   " .. col(DIM, "tick=" .. tostring(S.TickMode or "?")))

    local prefix, tailVal = parseStage(node.Stage)
    local minor = node.Minor or tailVal or 0
    local major = node.Major or 0
    local nextLabel
    if minor < 10 and prefix and prefix ~= "" then
        nextLabel = col(DIM, prefix .. " " .. (ROMAN[minor+1] or tostring(minor+1)))
    elseif minor >= 10 then
        nextLabel = col(WARN, "Major " .. (major+1) .. " · Minor 1  (new tier name not broadcast)")
    else
        nextLabel = col(DIM, "Major " .. major .. " · Minor " .. (minor+1))
    end
    table.insert(L, label("REALM") .. "  " .. hero(accent, node.Stage or "?")
        .. "   " .. col(DIM, "M" .. major .. "·m" .. minor) .. "  →  " .. nextLabel)

    local frac = (cost and cost > 0) and (cur / cost) or 0
    table.insert(L, label("ESSENCE") .. " " .. col(accent, fmtN(cur))
        .. col(DIM, "   cost ") .. col(HEAD, fmtN(cost)))
    table.insert(L, col(accent, bar(frac)) .. "  " .. heroSm(HEAD, string.format("%.1f%%", frac*100))
        .. "   " .. col(DIM, "gap ") .. col(HEAD, fmtN(gap)))

    table.insert(L, label("RATE") .. " " .. col(HEAD, fmtRate(rate)) .. col(DIM, " server")
        .. "   " .. col(DIM, fmtRate(obs.rate) .. " obs"))
    table.insert(L, label("TIME → COST") .. "  " .. hero(accent, fmtT(tCost)))

    table.insert(L, label("CHANCE") .. " now " .. col(HEAD, pct(chance))
        .. "   " .. col(DIM, "Δ " .. pctSlope(obs.cslope) .. " obs"))
    local chCostLine = label("CHANCE @ COST") .. " " .. col(GOOD, pct(chCost))
        .. col(DIM, "  (" .. modelName .. " fit, n=" .. (fit and fit.n or 0) .. ")")
    if vtx and vtx.concave and vtx.ch and vtx.ch < 1 then
        chCostLine = chCostLine .. col(WARN, "  peak " .. pct(vtx.ch))
    end
    table.insert(L, chCostLine)

    table.insert(L, label("ATTEMPT WHEN  ·  " .. modelName .. " model"))
    for _, T in ipairs({0.80, 0.90, 0.95, 0.99}) do
        local t, tag = attemptTimeFor(T, fit, curR, cur, cost, rate, chance)
        local rhs
        if t == nil then
            rhs = col(BAD, "—  ") .. col(DIM, "(" .. tostring(tag) .. ")")
        elseif tag == "attempt now" then
            rhs = col(GOOD, "now") .. col(DIM, "  (chance & essence both met)")
        elseif tag == "chance met now; gate = essence" then
            rhs = col(accent, fmtT(t)) .. col(DIM, "  (chance met now · gate = essence)")
        elseif tag == "ok@cost" then
            rhs = col(accent, fmtT(t)) .. col(DIM, "  (chance met by cost · gate = essence)")
        else
            rhs = col(accent, fmtT(t)) .. col(WARN, "  (overfill past cost)")
        end
        table.insert(L, "   ≥" .. math.floor(T*100) .. "%  →  " .. rhs)
    end
    if modelName == "linear" then
        table.insert(L, col("rgb(96,108,124)",
            "   linear slope can't see curve-bend near the top;"))
        table.insert(L, col("rgb(96,108,124)",
            "   high-% rows are lower bounds until the fit gains curvature."))
    else
        table.insert(L, col("rgb(96,108,124)",
            "   " .. modelName .. " fit over last " .. (fit and fit.n or 0) .. " ticks;"))
        table.insert(L, col("rgb(96,108,124)",
            "   values beyond the observed band are an estimate, not a guarantee."))
    end

    if node.BreakthroughWarning and node.BreakthroughWarning ~= "" then
        table.insert(L, col(WARN, "⚠ " .. node.BreakthroughWarning))
    end
    local lock = S.RealmLocks and S.RealmLocks[active]
    if lock then table.insert(L, col(BAD, "REALM LOCKED (" .. active .. ")")) end

    table.insert(L, "")
    table.insert(L, label("TECHNIQUES") .. col(DIM, "  active growth ×" .. tostring(node.TechniqueMultiplier)))
    local techs = S.Techniques
    if type(techs) == "table" then
        local keys = {}
        for k in pairs(techs) do table.insert(keys, k) end
        table.sort(keys, function(a,b) return tostring(a) < tostring(b) end)
        for _, k in ipairs(keys) do
            local t = techs[k]
            if type(t) == "table" then
                local flags = (t.affects_growth and "G" or "") .. (t.affects_recovery and "R" or "")
                local isActive = (t.path == active and t.affects_growth and t.multiplier == node.TechniqueMultiplier)
                local mark = isActive and col(GOOD, "  ◄ equipped") or ""
                local pcol = (t.path == "Qi") and QI or BODY
                table.insert(L, "   • " .. col(HEAD, tostring(t.name or "?"))
                    .. "  " .. col(pcol, "×" .. tostring(t.multiplier or "?"))
                    .. "  " .. col(DIM, "[" .. flags .. "]") .. mark)
            end
        end
    end

    local zq = S.ZoneMultiplierQi or 0
    local zb = S.ZoneMultiplierBody or 0
    local hereMult = (active == "Qi") and zq or zb
    local majQi   = (S.Qi and S.Qi.Major) or 0
    local majBody = (S.Body and S.Body.Major) or 0
    table.insert(L, "")
    table.insert(L, label("ZONES") .. col(DIM, "  in band ") .. heroSm(accent, fmtN(hereMult) .. "×")
        .. col(DIM, "   Qi " .. fmtN(zq) .. "× · Body " .. fmtN(zb) .. "×   [req≈Major]"))
    local ZONES = {
        {mult=5,req=0,path="Qi"},{mult=1000,req=1,path="All"},{mult=10000,req=6,path="Qi"},
        {mult=50000,req=10,path="Qi"},{mult=100000,req=15,path="Qi"},{mult=5000000,req=23,path="Qi"},
        {mult=5,req=0,path="Body"},{mult=10000,req=6,path="Body"},{mult=50000,req=10,path="Body"},
        {mult=100000,req=15,path="Body"},{mult=5000000,req=25,path="Body"},
    }
    local function qual(p, req)
        if p == "All" then return majQi >= req and majBody >= req end
        if p == "Qi" then return majQi >= req end
        return majBody >= req
    end
    local function zrow(z)
        local q = qual(z.path, z.req)
        local here = (z.mult == hereMult and (z.path == active or z.path == "All"))
        local arrow = here and col(accent, b("  ◄ you")) or ""
        local tag = (z.path == "All") and col(DIM, "all") or col(accent, z.path:lower())
        local mt = here and heroSm(accent, fmtN(z.mult) .. "×") or col(HEAD, fmtN(z.mult) .. "×")
        return "   " .. (q and col(GOOD, "✓") or col(DIM, "·")) .. " " .. mt
            .. col(DIM, "  req" .. z.req .. " ") .. tag .. arrow
    end
    for _, z in ipairs(ZONES) do if z.path == "All" then table.insert(L, zrow(z)) end end
    local actRows = {}
    for _, z in ipairs(ZONES) do if z.path == active then table.insert(actRows, z) end end
    table.sort(actRows, function(a,b) return a.mult < b.mult end)
    for _, z in ipairs(actRows) do table.insert(L, zrow(z)) end
    local oth = (active == "Qi") and "Body" or "Qi"
    local oAcc = (oth == "Qi") and QI or BODY
    local oNode = S[oth]
    local oParts = {}
    for _, z in ipairs(ZONES) do
        if z.path == oth then
            table.insert(oParts, (qual(z.path, z.req) and col(GOOD, "✓") or col(DIM, "·"))
                .. col(oAcc, fmtN(z.mult) .. "×") .. col(DIM, "(" .. z.req .. ")"))
        end
    end
    table.insert(L, col(DIM, "▹ " .. oth:upper()) .. "  "
        .. col(oAcc, oNode and (oNode.Stage or "?") or "?")
        .. "  " .. col(DIM, "chance ") .. col(HEAD, oNode and pct(oNode.BreakthroughChance) or "—")
        .. "   " .. table.concat(oParts, "  "))

    table.insert(L, "")
    table.insert(L, label("PROFILE"))
    table.insert(L, col(DIM, "Root ") .. col(HEAD, tostring(profile.root or "?"))
        .. col(DIM, "  Const ") .. col(HEAD, tostring(profile.const or "?"))
        .. col(DIM, "  Luck ") .. col(HEAD, tostring(profile.luck or "?"))
        .. col(DIM, "  Compr ") .. col(HEAD, tostring(profile.compr or "?")))
    table.insert(L, col(DIM, "Mastery  Forage ") .. col(HEAD, tostring(profile.mf or "?"))
        .. col(DIM, "  Alch ") .. col(HEAD, tostring(profile.ma or "?"))
        .. col(DIM, "  Weapon ") .. col(HEAD, tostring(profile.mw or "?"))
        .. col(DIM, "  Skill ") .. col(HEAD, tostring(profile.ms or "?")))
    local sp = S.Spirit
    table.insert(L, col(DIM, "SpiritStones ") .. col(HEAD, tostring(profile.ss or "?"))
        .. col(DIM, "   Spirit ") .. col(HEAD, (sp and tostring(sp.Current) or "?") .. "/" .. (sp and tostring(sp.Max) or "?")))

    bigLabel.Text = table.concat(L, "\n")
    pcall(function() scroll.CanvasSize = UDim2.new(0,0,0, bigLabel.AbsoluteSize.Y + 24) end)

    -- v2.4: living auto-bt pill + feedback line
    if autoBtPill then
        local liveCh = nil
        local ra = normPath(S.ActivePath or S.GainPath)
        if ra and S[ra] then liveCh = S[ra].BreakthroughChance end
        if SET.autoBt then
            local chTxt = liveCh and (" · now " .. pct(liveCh)) or ""
            autoBtPill.Text = "AUTO-BT  ON   " .. (ra or active) .. "≥100%"
                .. (SET.autoBtBody and "  +BODY⚠" or "") .. chTxt
            autoBtPill.BackgroundColor3 = (tick() < btFlashUntil) and rgb(70,170,100) or rgb(40,96,64)
            autoBtPill.TextColor3 = rgb(225,255,235)
        else
            autoBtPill.Text = "AUTO-BT  OFF   (tap → arm Qi at 100%)"
            autoBtPill.BackgroundColor3 = rgb(46,51,66)
            autoBtPill.TextColor3 = rgb(150,162,178)
        end
    end
    if btFeedbackLabel then
        btFeedbackLabel.Text = btFeedback
        btFeedbackLabel.TextColor3 = (tick() < btFlashUntil) and C.good or C.inkDim
    end
end

local function mk(cls, props, parent)
    local o = Instance.new(cls)
    for k,v in pairs(props) do o[k] = v end
    if parent then o.Parent = parent end
    return o
end

local function makeVeil(parent)
    -- v2.5: rest state fully transparent + never Active → can NEVER block clicks
    -- or hide content even if a tween misbehaves.
    local v = mk("Frame", {
        Size = UDim2.new(1,0,1,0), BackgroundColor3 = C.field,
        BorderSizePixel = 0, Transparency = 1, Visible = false,
        ZIndex = 9, Active = false,
    }, parent)
    return v
end
-- v2.5: NON-BLOCKING, SELF-HEALING page sweep. Never opaque at rest; a soft
-- tint that fades OUT; a hard safety timer force-clears it no matter what.
local function pageIn(veil)
    if not veil then return end
    veil.Active = false
    veil.Transparency = 0.5
    veil.Visible = true
    local t = tw(veil, {Transparency = 1}, 0.28)
    if t then
        task.spawn(function()
            pcall(function() t.Completed:Wait() end)
            veil.Visible = false; veil.Transparency = 1
        end)
    end
    task.delay(0.45, function()   -- safety net: the veil can never stick as a sheet
        if veil then veil.Visible = false; veil.Transparency = 1 end
    end)
end
local function getGuiParent()
    local ok, r
    if gethui then ok, r = pcall(gethui); if ok and r then return r end end
    ok, r = pcall(function() return game:GetService("CoreGui") end)
    if ok and r then return r end
    ok, r = pcall(function() return LP:WaitForChild("PlayerGui") end)
    return ok and r or nil
end
local function destroyExistingGui()
    pcall(function()
        local parents = {}
        if gethui then local ok,r = pcall(gethui); if ok and r then table.insert(parents, r) end end
        pcall(function() table.insert(parents, game:GetService("CoreGui")) end)
        pcall(function() local pg = LP:FindFirstChildOfClass("PlayerGui"); if pg then table.insert(parents, pg) end end)
        for _, p in ipairs(parents) do
            for _, g in ipairs(p:GetChildren()) do
                if g.Name == "JianghuCultENI" then g:Destroy() end
            end
        end
    end)
end

local function serializeVal(val, depth, seen)
    depth = depth or 3; seen = seen or {}
    local t = typeof(val)
    if t == "nil" then return "nil"
    elseif t == "boolean" or t == "number" then return tostring(val)
    elseif t == "string" then
        if #val > 120 then return string.format("%q...", val:sub(1,120)) end
        return string.format("%q", val)
    elseif t == "Instance" then
        local ok, p = pcall(function() return val:GetFullName() end)
        return ok and p or val.ClassName
    elseif t == "table" then
        if depth <= 0 then return "{...}" end
        if seen[val] then return "{<cycle>}" end
        seen[val] = true
        local parts, n = {}, 0
        for k, v in pairs(val) do
            n = n + 1
            if n > 24 then table.insert(parts, "...+" .. (n-24)) break end
            table.insert(parts, tostring(k) .. "=" .. serializeVal(v, depth-1, seen))
        end
        return "{" .. table.concat(parts, ", ") .. "}"
    else return string.format("[%s]", t)
    end
end

local function doBreakthrough(path)
    local r = getRemote("Network.Cultivation.breakthrough")
    if not r then
        btFeedback = "no breakthrough remote found"
        return
    end
    pcall(function() r:FireServer(path) end)
    autoBtLastFire = tick()
    btAwaiting = true
    btAwaitPath = path
    btAwaitStage = (S and S[path] and S[path].Stage) or "?"
    btFlashUntil = tick() + 1.2
    local chNow = (S and S[path] and S[path].BreakthroughChance) or 0
    btFeedback = "fired " .. path .. " @ " .. pct(chNow) .. " · cooldown " .. AUTOBT_COOLDOWN .. "s"
end

local function autoBtLoop()
    task.spawn(function()
        while running do
            task.wait(0.5)
            if SET.autoBt and S
               and (tick() - autoBtBootT) >= AUTOBT_BOOT_GRACE
               and (tick() - autoBtLastFire) >= AUTOBT_COOLDOWN then
                local realActive = normPath(S.ActivePath or S.GainPath)
                if realActive then
                    local node = S[realActive]
                    if node then
                        local allowed = (realActive == "Qi")
                                     or (realActive == "Body" and SET.autoBtBody)
                        if allowed and node.CanBreakthrough then
                            local cur  = node.Current or 0
                            local cost = node.BreakthroughCost or 0
                            local ch   = node.BreakthroughChance or 0
                            if cost > 0 and cur >= cost and ch >= (SET.minChance / 100) then
                                doBreakthrough(realActive)
                            end
                        end
                    end
                end
            end
            if btAwaiting and S and btAwaitPath then
                local nodeNow = S[btAwaitPath]
                if nodeNow then
                    if nodeNow.Stage and nodeNow.Stage ~= btAwaitStage then
                        btFeedback = "✓ REALM UP → " .. tostring(nodeNow.Stage)
                        btFlashUntil = tick() + 2.0
                        btAwaiting = false
                    elseif (tick() - autoBtLastFire) > 8 then
                        btFeedback = "roll resolved · essence reset · re-grinding"
                        btAwaiting = false
                    end
                end
            end
        end
    end)
end

local function startLoops()
    task.spawn(function()
        local pacc = 0
        while running do
            task.wait(0.2)
            pacc = pacc + 0.2
            if pacc >= 2 then pacc = 0; pcall(pollProfile) end
            if page == "main" and not minimized then pcall(redraw) end
        end
    end)
    autoBtLoop()   -- v2.4
end
